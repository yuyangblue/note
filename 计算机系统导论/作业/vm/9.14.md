#include <sys/mman.h>

#include <sys/stat.h>

#include <fcntl.h>

#include <unistd.h>

#include <string.h>

#include <stdio.h>

  

int main(){

    int fd = open("hello.txt", O_RDWR);

    if(fd == -1){

        perror("open");

        return 1;

    }

  

    struct stat st;

    fstat(fd, &st);

    size_t len = st.st_size;

  

    char *p = mmap(NULL, len,

    PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);

    if(p == MAP_FAILED){

        perror("mmap");

        return 1;

    }

  

    memcpy(p, "Jello, world!", 13);

    munmap(p, len);

    close(fd);

    return 0;

}

#include "csapp.h"

int main(int argc, char **argv, bool infile){
	int n;
	rio_t rio;
	char buf[MAXLINE];
	
	if(infile){
		Rio_readinitb(&rio, argv[1]);
	}else Rio_readinitb(&rio, STDIN_FILENO);
	while((n = Rio_readlineb(&rio, buf, MAXLINE)) != 0)
	Rio_writen(STDOUT_FILENO, buf, n);
	exit(0);
}

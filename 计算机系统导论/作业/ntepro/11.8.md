void sigchld_handler(int sig)

{

    int olderrno = errno;                   /* 保存被打断代码的 errno */

    (void)sig;                              /* 未使用参数，显式忽略 */

  

    while (waitpid(-1, NULL, WNOHANG) > 0)

        ;   /* -1：回收任意子进程；NULL：不关心退出状态；

               WNOHANG：没有已终止的子进程时立即返回 0 而不阻塞 */

  

    errno = olderrno;                       /* 恢复 errno */

}

    void serve_dynamic(int fd, char *filename, char *cgiargs, int is_head)

{

    char buf[MAXLINE], *emptylist[] = { NULL };

  

    /* 先发响应头的前半部分（200 状态行 + Server），头部到此为止 */

    sprintf(buf, "HTTP/1.0 200 OK\r\n");

    Rio_writen(fd, buf, strlen(buf));

    sprintf(buf, "Server: Tiny Web Server\r\n");

    Rio_writen(fd, buf, strlen(buf));

  

    /* HEAD 请求：不再 fork/execve 运行 CGI，响应到此结束 */

    if (is_head)

        return;

  

    if (Fork() == 0) { /* 子进程：负责跑 CGI 程序 */ //line:netp:servedynamic:fork

    /* 真实的 Web 服务器还会在这里设置其余全部 CGI 环境变量 */

    setenv("QUERY_STRING", cgiargs, 1); /* 传入查询串，如 "3&4" */ //line:netp:servedynamic:setenv

    Dup2(fd, STDOUT_FILENO);         /* 标准输出重定向到客户端套接字 */ //line:netp:servedynamic:dup2

    Execve(filename, emptylist, environ); /* 加载并运行 CGI 程序 */ //line:netp:servedynamic:execve

    }

    /* 习题 11.8：原版在这里调用 Wait(NULL) 阻塞等待子进程终止并回收。

     * 现在改为【不等待】：父进程立即返回继续处理下一个连接，

     * 僵尸进程由 SIGCHLD 处理程序（sigchld_handler）异步回收。 */

    // Wait(NULL); /* 已删除：显式等待改为 SIGCHLD 异步回收 */

}

void doit(int fd)

{

    int is_static;

    int is_head;             /* 习题：请求方法是 HEAD 时置 1，只发头不发体 */

    struct stat sbuf;

    char buf[MAXLINE], method[MAXLINE], uri[MAXLINE], version[MAXLINE];

    char filename[MAXLINE], cgiargs[MAXLINE];

    // char reqline[MAXLINE];             // 作业 A：保存原始请求行用于回显（已停用）

    rio_t rio;

  

    /* 1. 用 csapp 的 Robust I/O（缓冲读写）初始化，并把连接 fd 绑定上去 */

    Rio_readinitb(&rio, fd);

  

    /* 2. 读请求行，例如 "GET /home.html HTTP/1.0\r\n"

     *    对端关闭连接时返回 0，直接结束本次事务 */

    if (!Rio_readlineb(&rio, buf, MAXLINE))  //line:netp:doit:readrequest

        return;

    printf("%s", buf);                        /* 请求行打到服务器日志 */

  

    /* 3. 按空白把请求行切成三个字段：方法、URI、协议版本 */

    sscanf(buf, "%s %s %s", method, uri, version);       //line:netp:doit:parserequest

    /* 【习题】接受 GET 和 HEAD 两种方法；其他方法返回 501 Not Implemented。

     * strcasecmp 相等返回 0，所以"既不是 GET 也不是 HEAD"时两个调用

     * 都非 0，&& 条件成立 */

    if (strcasecmp(method, "GET") && strcasecmp(method, "HEAD")) { //line:netp:doit:beginrequesterr

        clienterror(fd, method, "501", "Not Implemented",

                    "Tiny does not implement this method");

        return;

    }                                                    //line:netp:doit:endrequesterr

    is_head = !strcasecmp(method, "HEAD"); /* HEAD 置 1；GET 时为 0 */

  

    /* ===== 作业 A：把请求行和请求报头原样回显给客户端 =====

     * 【已注释停用】当前为原版行为（正常提供文件/CGI 服务）。

     * 若要启用回显模式：取消本段注释，并恢复 read_requesthdrs

     * 里被注释的两行 Rio_writen。 */

    /*

    strcpy(reqline, buf);                // 先备份原始请求行

    sprintf(buf, "HTTP/1.0 200 OK\r\n"); // 响应头：状态行

    Rio_writen(fd, buf, strlen(buf));

    sprintf(buf, "Server: Tiny Web Server\r\n");

    Rio_writen(fd, buf, strlen(buf));

    sprintf(buf, "Content-type: text/plain\r\n\r\n");

    Rio_writen(fd, buf, strlen(buf));

    Rio_writen(fd, reqline, strlen(reqline)); // 回显请求行，原样含 \r\n

  

    // 回显完本次事务即结束，不再发送静态/动态内容

    return;

    */

  

    /* 4. GET 没有请求体，头部行读出来只为打印日志，读完忽略 */

    read_requesthdrs(&rio, fd);                          //line:netp:doit:readrequesthdrs

  

    /* 5. 解析 URI：

     *    静态 -> filename = "./home.html" 之类，cgiargs 为空

     *    动态 -> filename = "./cgi-bin/adder"，cgiargs = "?后面的参数"

     *    返回 1 表示静态，0 表示动态 */

    is_static = parse_uri(uri, filename, cgiargs);       //line:netp:doit:staticcheck

  

    /* 6. 文件必须存在，否则 404（注意：这里查的是磁盘文件/CGI 可执行文件） */

    if (stat(filename, &sbuf) < 0) {                     //line:netp:doit:beginnotfound

    clienterror(fd, filename, "404", "Not found",

            "Tiny couldn't find this file");

    return;

    }                                                    //line:netp:doit:endnotfound

  

    if (is_static) { /* 静态内容：必须是普通文件且拥有者有读权限 */

    if (!(S_ISREG(sbuf.st_mode)) || !(S_IRUSR & sbuf.st_mode)) { //line:netp:doit:readable

        clienterror(fd, filename, "403", "Forbidden",

            "Tiny couldn't read the file");

        return;

    }

    serve_static(fd, filename, sbuf.st_size, is_head); //line:netp:doit:servestatic

    }

    else { /* 动态内容：必须是普通文件且拥有者有执行权限 */

    if (!(S_ISREG(sbuf.st_mode)) || !(S_IXUSR & sbuf.st_mode)) { //line:netp:doit:executable

        clienterror(fd, filename, "403", "Forbidden",

            "Tiny couldn't run the CGI program");

        return;

    }

    serve_dynamic(fd, filename, cgiargs, is_head);   //line:netp:doit:servedynamic

    }

}

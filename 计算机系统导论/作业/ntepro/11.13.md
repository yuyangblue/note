static void safe_writen(int fd, void *usrbuf, size_t n)
{
    if (rio_writen(fd, usrbuf, n) != (ssize_t)n) {
        if (errno == EPIPE)
            printf("对端过早关闭连接（EPIPE），放弃写入剩余数据\n");
        else
            fprintf(stderr, "write: %s\n", strerror(errno));
        /* 关键：不 exit、不 return -1 给调用者 ... */
    }
}
#define Rio_writen(f, p, len) safe_writen((f), (p), (len))

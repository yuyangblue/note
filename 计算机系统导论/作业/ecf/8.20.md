```c
#include <stdio.h>
#include <unistd.h>
extern char **environ;   /* 由 C 运行库（libc）维护的全局变量，声明在 <unistd.h> */
int main(int argc, char *argv[]){
    argv[0] = "ls";
    execve("/bin/ls", argv, environ);
    // 能执行到这里一定是execve失败
    perror("execve fail");
    return 1;
}
```

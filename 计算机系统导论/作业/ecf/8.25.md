```c
#include "csapp.h"
#define TIMEOUT 5
static sigjmp_buf env;   /* 保存跳转环境 */
void sigalrm_handler(int sig)   /* SIGALRM 处理程序 */
{
    siglongjmp(env, 1);   /* 非本地跳转：飞回 sigsetjmp 处，返回值 1 */
}
char *tfgets(char *buf, int bufsize, FILE *stream)
{
    if (sigsetjmp(env, 1) != 0)   /* 超时后跳回这里，返回值 1 */
        return NULL;              /* 超时 → 返回 NULL */
    if (signal(SIGALRM, sigalrm_handler) == SIG_ERR)
        unix_error("set alarm handler error");
    alarm(TIMEOUT);               /* 5 秒后向本进程发 SIGALRM */
    return fgets(buf, bufsize, stream);   /* 正常输入 → 返回行指针 */
}
```

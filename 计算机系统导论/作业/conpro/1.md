 编写一个hello.c（下图）的一个版本，它创建和回收n个可结合的对等线程，其中n是一个命令行参数。
![[Pasted image 20261126202028.png|820]]

```c
#include "csapp.h"

void *thread(void *vargp);
int main(int argc, char **argv)

{

    if (argc != 2)

        return 1;

    int n = atoi(argv[1]);

    if (n <= 0)

        return 1;

    pthread_t *tids = Malloc(n * sizeof(pthread_t));

    for (int i = 0; i < n; i++)

        Pthread_create(&tids[i], NULL, thread, NULL);

    for (int i = 0; i < n; i++)

        Pthread_join(tids[i], NULL);

    Free(tids);

    return 0;

}


void *thread(void *vargp)

{

    (void)vargp;

    printf("Hello, world\n");

    return NULL;

}
```

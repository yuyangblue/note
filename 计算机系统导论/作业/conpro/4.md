```c
#include "csapp.h"



int echo_one_line(int connfd, rio_t *rio);

void command(void);



int main(int argc, char **argv)

{

    int listenfd, connfd = -1, maxfd;

    socklen_t clientlen = sizeof(struct sockaddr_in);

    struct sockaddr_in clientaddr;

    fd_set read_set, ready_set;

    static rio_t rio;



    if (argc != 2) {

        fprintf(stderr, "usage: %s <port>\n", argv[0]);

        exit(0);

    }

    listenfd = Open_listenfd(argv[1]);

    maxfd = listenfd;



    FD_ZERO(&read_set);

    FD_SET(STDIN_FILENO, &read_set);

    FD_SET(listenfd, &read_set);



    while (1) {

        ready_set = read_set;

        Select(maxfd+1, &ready_set, NULL, NULL, NULL);

        if (FD_ISSET(STDIN_FILENO, &ready_set))

            command();              /* read command line from stdin */

        if (FD_ISSET(listenfd, &ready_set)) {

            connfd = Accept(listenfd, (SA *)&clientaddr, &clientlen);

            clientlen = sizeof(struct sockaddr_in);

            Rio_readinitb(&rio, connfd);

            FD_SET(connfd, &read_set);

            if(connfd > maxfd){

                maxfd = connfd;

            }

        }



        if(connfd != -1 && FD_ISSET(connfd, &ready_set)){

            if(echo_one_line(connfd, &rio) == 0){

                Close(connfd);

                FD_CLR(connfd, &read_set);

                connfd = -1;

                maxfd = listenfd;

            }

        }

    }

}



int echo_one_line(int connfd, rio_t *rio){

    char buf[MAXLINE];

    ssize_t n = Rio_readlineb(rio, buf, MAXLINE);

    if(n == 0)

        return 0;

    Rio_writen(connfd, buf, n);

    return 1;

}



void command(void)

{

    char buf[MAXLINE];

    if (!fgets(buf, MAXLINE, stdin))

        exit(0);                    /* EOF */

    printf("%s", buf);              /* Process the input command */

}
```

**改了哪些地方（相对教材原版的 select.c）**

- 把 `connfd` 也 `FD_SET` 进 `read_set`，它才在 select 的监视范围内；否则连接上的数据永远不会唤醒服务器。
- accept 后更新 `maxfd`。`Select` 的第一个参数是「最大 fd + 1」，不更新就会漏查 `connfd`（它通常比 `listenfd` 大）。
- 连接可读时只调一次 `echo_one_line`：`Rio_readlineb` 读一行、`Rio_writen` 回一行就返回，不进 `echo` 的读循环，所以每轮迭代最多回送一行。
- 读到 EOF（返回 0）时 `Close` + `FD_CLR` + `connfd = -1` 三件事都要做：少 `FD_CLR` 会让 select 一直报这个已关闭的 fd 可读，程序空转。
- `rio` 声明成 `static`：`Rio_readlineb` 带缓冲，一次 read 可能读进好几行，缓冲区必须跨迭代保留，不能每轮重新 `Rio_readinitb`。

**两个已知局限**

- 只支持一个连接：第二个客户端连上会覆盖 `connfd`，旧 fd 仍留在 `read_set` 里，select 会一直报它可读、程序却只查新的 `connfd`，于是空转。要并发多连接得用图 12-8 的 pool（`clientfd[]` + `clientrio[]`）。
- `Rio_readlineb` 要凑满一整行才返回，客户端只发半行就会阻塞整个服务器——这正是练习题 12.23 要解决的问题。

    1. 大白话引入¶

    想象你有一本 1000 页的书，老师让你快速找到第 300-700 页里有多少个"算法"这个词。

    暴力做法：一页一页翻，从第 300 页翻到第 700 页。太慢了。

    分块思想：把书分成 30 块，每块约 33 页。提前统计好每块里有多少个"算法"。查询时： - 中间的整块（第 10-21 块）：直接查预处理的答案，O(1) - 两端的散块（第 9 块的一部分、第 22 块的一部分）：暴力数

    这样查询复杂度从 O(区间长度) 降到 O(块数 + 块大小) = O(√N)。

    分块 vs 线段树： - 线段树：递归结构，log N，但需要信息可合并 - 分块：两层结构（整块 + 散块），√N，但不需要合并，能维护更复杂的信息

    2. 基础分块：区间加、单点查询、区间和¶

    核心思想¶

    把长度为 N 的序列分成 √N 个块，每块大小 B = √N。

    区间加操作： - 整块：打懒标记 add[i]，表示第 i 块所有数都要加上这个值，O(1) - 散块：暴力修改原数组，O(B)

    单点查询： - 原数组值 + 所在块的懒标记

    区间和查询： - 整块：预处理块的和 sum[i]，查询时 sum[i] + add[i] * B - 散块：暴力累加 a[j] + add[块]

    #include <bits/stdc++.h>
using namespace std;

const int MAXN = 100005;
int a[MAXN];           // 原数组
long long sum[MAXN];   // 每块的和
long long add[MAXN];   // 每块的懒标记
int bl[MAXN];          // bl[i] 表示第 i 个元素属于第几块
int L[MAXN], R[MAXN];  // 每块的左右端点
int n, m, B;           // B 是块大小

// 初始化分块
void init() {
    B = sqrt(n);
    int num = (n + B - 1) / B;  // 块数
    for (int i = 1; i <= num; i++) {
        L[i] = (i - 1) * B + 1;
        R[i] = min(i * B, n);
    }
    for (int i = 1; i <= n; i++) {
        bl[i] = (i - 1) / B + 1;
        sum[bl[i]] += a[i];
    }
}

// 区间加：给 [l, r] 加上 v
void range_add(int l, int r, int v) {
    int bl_l = bl[l], bl_r = bl[r];

    // 如果在同一块，暴力修改
    if (bl_l == bl_r) {
        for (int i = l; i <= r; i++) {
            a[i] += v;
            sum[bl_l] += v;
        }
        return;
    }

    // 左散块：暴力修改
    for (int i = l; i <= R[bl_l]; i++) {
        a[i] += v;
        sum[bl_l] += v;
    }

    // 中间整块：打懒标记
    for (int i = bl_l + 1; i < bl_r; i++) {
        add[i] += v;
    }

    // 右散块：暴力修改
    for (int i = L[bl_r]; i <= r; i++) {
        a[i] += v;
        sum[bl_r] += v;
    }
}

// 单点查询：查询位置 pos 的值
long long query_point(int pos) {
    return a[pos] + add[bl[pos]];
}

// 区间和查询：查询 [l, r] 的和
long long query_sum(int l, int r) {
    int bl_l = bl[l], bl_r = bl[r];
    long long ans = 0;

    // 如果在同一块，暴力累加
    if (bl_l == bl_r) {
        for (int i = l; i <= r; i++) {
            ans += a[i] + add[bl_l];
        }
        return ans;
    }

    // 左散块
    for (int i = l; i <= R[bl_l]; i++) {
        ans += a[i] + add[bl_l];
    }

    // 中间整块
    for (int i = bl_l + 1; i < bl_r; i++) {
        ans += sum[i] + add[i] * (R[i] - L[i] + 1);
    }

    // 右散块
    for (int i = L[bl_r]; i <= r; i++) {
        ans += a[i] + add[bl_r];
    }

    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
    }

    init();

    while (m--) {
        int op, l, r;
        cin >> op >> l >> r;
        if (op == 1) {  // 区间加
            int v;
            cin >> v;
            range_add(l, r, v);
        } else {  // 区间和查询
            cout << query_sum(l, r) << "\n";
        }
    }

    return 0;
}

    复杂度分析： - 区间加：O(B + N/B)，当 B = √N 时最优，O(√N) - 单点查询：O(1) - 区间和查询：O(B + N/B) = O(√N)

    3. 进阶：区间乘与标记下放¶

    问题¶

    现在要支持两种操作： 1. 区间乘：给 [l, r] 所有数乘上 v 2. 区间加：给 [l, r] 所有数加上 v

    核心思想¶

    每个块维护两个标记：mul[i]（乘法标记）和 add[i]（加法标记）。

    块内元素的实际值 = a[i] * mul[块] + add[块]

    区间乘 v： - 整块：mul[i] *= v，add[i] *= v - 散块：先下放标记（把标记作用到原数组），再暴力修改

    区间加 v： - 整块：add[i] += v - 散块：先下放标记，再暴力修改

    标记下放：把 mul 和 add 作用到原数组，然后重置标记。

    // 区间乘+区间加的标记下放与操作（思路示意，完整实现见基础分块章节）
#include <bits/stdc++.h>
using namespace std;

const int MAXN = 100005;
long long a[MAXN];
long long sum[MAXN];
long long mul[MAXN], add[MAXN];
int bl[MAXN];
int L[MAXN], R[MAXN];
int n, B;

// 下放第 b 块的标记
void push_down(int b) {
    if (mul[b] != 1 || add[b] != 0) {
        for (int i = L[b]; i <= R[b]; i++) {
            a[i] = a[i] * mul[b] + add[b];
        }
        mul[b] = 1;
        add[b] = 0;
    }
}

// 区间乘
void range_mul(int l, int r, int v) {
    int bl_l = bl[l], bl_r = bl[r];

    if (bl_l == bl_r) {
        push_down(bl_l);
        for (int i = l; i <= r; i++) {
            a[i] *= v;
        }
        sum[bl_l] = 0;
        for (int i = L[bl_l]; i <= R[bl_l]; i++) {
            sum[bl_l] += a[i];
        }
        return;
    }

    push_down(bl_l);
    for (int i = l; i <= R[bl_l]; i++) {
        a[i] *= v;
    }
    sum[bl_l] = 0;
    for (int i = L[bl_l]; i <= R[bl_l]; i++) {
        sum[bl_l] += a[i];
    }

    for (int i = bl_l + 1; i < bl_r; i++) {
        mul[i] *= v;
        add[i] *= v;
        sum[i] *= v;
    }

    push_down(bl_r);
    for (int i = L[bl_r]; i <= r; i++) {
        a[i] *= v;
    }
    sum[bl_r] = 0;
    for (int i = L[bl_r]; i <= R[bl_r]; i++) {
        sum[bl_r] += a[i];
    }
}

// 区间加
void range_add(int l, int r, int v) {
    int bl_l = bl[l], bl_r = bl[r];

    if (bl_l == bl_r) {
        push_down(bl_l);
        for (int i = l; i <= r; i++) {
            a[i] += v;
        }
        sum[bl_l] = 0;
        for (int i = L[bl_l]; i <= R[bl_l]; i++) {
            sum[bl_l] += a[i];
        }
        return;
    }

    push_down(bl_l);
    for (int i = l; i <= R[bl_l]; i++) {
        a[i] += v;
    }
    sum[bl_l] = 0;
    for (int i = L[bl_l]; i <= R[bl_l]; i++) {
        sum[bl_l] += a[i];
    }

    for (int i = bl_l + 1; i < bl_r; i++) {
        add[i] += v;
        sum[i] += v * (R[i] - L[i] + 1);
    }

    push_down(bl_r);
    for (int i = L[bl_r]; i <= r; i++) {
        a[i] += v;
    }
    sum[bl_r] = 0;
    for (int i = L[bl_r]; i <= R[bl_r]; i++) {
        sum[bl_r] += a[i];
    }
}

int main() { return 0; }

    #include <bits/stdc++.h>
using namespace std;

const int MAXN = 100005;
long long a[MAXN];
long long sum[MAXN];
int cnt[MAXN];  // 开方次数
int bl[MAXN];
int L[MAXN], R[MAXN];
int n, m, B;

void init() {
    B = sqrt(n);
    int num = (n + B - 1) / B;
    for (int i = 1; i <= num; i++) {
        L[i] = (i - 1) * B + 1;
        R[i] = min(i * B, n);
    }
    for (int i = 1; i <= n; i++) {
        bl[i] = (i - 1) / B + 1;
        sum[bl[i]] += a[i];
    }
}

void range_sqrt(int l, int r) {
    int bl_l = bl[l], bl_r = bl[r];

    if (bl_l == bl_r) {
        for (int i = l; i <= r; i++) {
            sum[bl_l] -= a[i];
            a[i] = sqrt(a[i]);
            sum[bl_l] += a[i];
        }
        return;
    }

    // 左散块
    for (int i = l; i <= R[bl_l]; i++) {
        sum[bl_l] -= a[i];
        a[i] = sqrt(a[i]);
        sum[bl_l] += a[i];
    }

    // 中间整块
    for (int i = bl_l + 1; i < bl_r; i++) {
        if (cnt[i] >= 5) continue;  // 已经全是 1，跳过
        cnt[i]++;
        sum[i] = 0;
        for (int j = L[i]; j <= R[i]; j++) {
            a[j] = sqrt(a[j]);
            sum[i] += a[j];
        }
    }

    // 右散块
    for (int i = L[bl_r]; i <= r; i++) {
        sum[bl_r] -= a[i];
        a[i] = sqrt(a[i]);
        sum[bl_r] += a[i];
    }
}

long long query_sum(int l, int r) {
    int bl_l = bl[l], bl_r = bl[r];
    long long ans = 0;

    if (bl_l == bl_r) {
        for (int i = l; i <= r; i++) {
            ans += a[i];
        }
        return ans;
    }

    for (int i = l; i <= R[bl_l]; i++) {
        ans += a[i];
    }
    for (int i = bl_l + 1; i < bl_r; i++) {
        ans += sum[i];
    }
    for (int i = L[bl_r]; i <= r; i++) {
        ans += a[i];
    }

    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
    }

    init();

    while (m--) {
        int op, l, r;
        cin >> op >> l >> r;
        if (op == 1) {
            range_sqrt(l, r);
        } else {
            cout << query_sum(l, r) << "\n";
        }
    }

    return 0;
}

using namespace std;

const int MAXN = 100005;
int a[MAXN];
int tag[MAXN];  // tag[i] = -1 表示没有统一颜色
int bl[MAXN];
int L[MAXN], R[MAXN];
int n, m, B;

void init() {
    B = sqrt(n);
    int num = (n + B - 1) / B;
    for (int i = 1; i <= num; i++) {
        L[i] = (i - 1) * B + 1;
        R[i] = min(i * B, n);
        tag[i] = -1;
    }
    for (int i = 1; i <= n; i++) {
        bl[i] = (i - 1) / B + 1;
    }
}

// 下放标记
void push_down(int b) {
    if (tag[b] != -1) {
        for (int i = L[b]; i <= R[b]; i++) {
            a[i] = tag[b];
        }
        tag[b] = -1;
    }
}

// 查询 [l, r] 中 c 的个数
int query_count(int l, int r, int c) {
    int bl_l = bl[l], bl_r = bl[r];
    int ans = 0;

    if (bl_l == bl_r) {
        push_down(bl_l);
        for (int i = l; i <= r; i++) {
            if (a[i] == c) ans++;
        }
        return ans;
    }

    push_down(bl_l);
    for (int i = l; i <= R[bl_l]; i++) {
        if (a[i] == c) ans++;
    }

    for (int i = bl_l + 1; i < bl_r; i++) {
        if (tag[i] == c) {
            ans += R[i] - L[i] + 1;
        } else if (tag[i] == -1) {
            for (int j = L[i]; j <= R[i]; j++) {
                if (a[j] == c) ans++;
            }
        }
    }

    push_down(bl_r);
    for (int i = L[bl_r]; i <= r; i++) {
        if (a[i] == c) ans++;
    }

    return ans;
}

// 把 [l, r] 改成 c
void range_assign(int l, int r, int c) {
    int bl_l = bl[l], bl_r = bl[r];

    if (bl_l == bl_r) {
        push_down(bl_l);
        for (int i = l; i <= r; i++) {
            a[i] = c;
        }
        return;
    }

    push_down(bl_l);
    for (int i = l; i <= R[bl_l]; i++) {
        a[i] = c;
    }

    for (int i = bl_l + 1; i < bl_r; i++) {
        tag[i] = c;
    }

    push_down(bl_r);
    for (int i = L[bl_r]; i <= r; i++) {
        a[i] = c;
    }
}

        } else {

    #include <bits/stdc++.h>
using namespace std;

const int MAXN = 100005;
int a[MAXN];
vector<int> sorted_blk[MAXN / 300 + 5];  // 每块的排序副本
long long add[MAXN / 300 + 5];           // 懒标记
int bl[MAXN];
int L[MAXN], R[MAXN];
int n, m, B;
int num_blk;

void init() {
    B = sqrt(n);
    num_blk = (n + B - 1) / B;
    for (int i = 1; i <= num_blk; i++) {
        L[i] = (i - 1) * B + 1;
        R[i] = min(i * B, n);
        sorted_blk[i].clear();
        for (int j = L[i]; j <= R[i]; j++) {
            bl[j] = i;
            sorted_blk[i].push_back(a[j]);
        }
        sort(sorted_blk[i].begin(), sorted_blk[i].end());
    }
}

// 重建第 b 块的排序副本
void rebuild(int b) {
    sorted_blk[b].clear();
    for (int i = L[b]; i <= R[b]; i++) {
        sorted_blk[b].push_back(a[i] + add[b]);
    }
    sort(sorted_blk[b].begin(), sorted_blk[b].end());
}

// 区间加
void range_add(int l, int r, int v) {
    int bl_l = bl[l], bl_r = bl[r];

    if (bl_l == bl_r) {
        for (int i = l; i <= r; i++) {
            a[i] += v;
        }
        rebuild(bl_l);
        return;
    }

    for (int i = l; i <= R[bl_l]; i++) {
        a[i] += v;
    }
    rebuild(bl_l);

    for (int i = bl_l + 1; i < bl_r; i++) {
        add[i] += v;
    }

    for (int i = L[bl_r]; i <= r; i++) {
        a[i] += v;
    }
    rebuild(bl_r);
}

// 查询 [l, r] 中小于 x 的个数
int query_less(int l, int r, long long x) {
    int bl_l = bl[l], bl_r = bl[r];
    int ans = 0;

    if (bl_l == bl_r) {
        for (int i = l; i <= r; i++) {
            if (a[i] + add[bl_l] < x) ans++;
        }
        return ans;
    }

    for (int i = l; i <= R[bl_l]; i++) {
        if (a[i] + add[bl_l] < x) ans++;
    }

    for (int i = bl_l + 1; i < bl_r; i++) {
        // 在 sorted_blk[i] 中二分查找 x - add[i]
        int cnt = lower_bound(sorted_blk[i].begin(), sorted_blk[i].end(), x - add[i]) - sorted_blk[i].begin();
        ans += cnt;
    }

    for (int i = L[bl_r]; i <= r; i++) {
        if (a[i] + add[bl_r] < x) ans++;
    }

    return ans;
}

// 查询 [l, r] 中 x 的前驱
long long query_pred(int l, int r, long long x) {
    int bl_l = bl[l], bl_r = bl[r];
    long long ans = -1e18;

    if (bl_l == bl_r) {
        for (int i = l; i <= r; i++) {
            long long val = a[i] + add[bl_l];
            if (val < x) ans = max(ans, val);
        }
        return ans;
    }

    for (int i = l; i <= R[bl_l]; i++) {
        long long val = a[i] + add[bl_l];
        if (val < x) ans = max(ans, val);
    }

    for (int i = bl_l + 1; i < bl_r; i++) {
        // 在 sorted_blk[i] 中找小于 x - add[i] 的最大值
        auto it = lower_bound(sorted_blk[i].begin(), sorted_blk[i].end(), x - add[i]);
        if (it != sorted_blk[i].begin()) {
            --it;
            ans = max(ans, *it + add[i]);
        }
    }

    for (int i = L[bl_r]; i <= r; i++) {
        long long val = a[i] + add[bl_r];
        if (val < x) ans = max(ans, val);
    }

    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
    }

    init();

    while (m--) {
        int op;
        cin >> op;
        if (op == 1) {
            int l, r, v;
            cin >> l >> r >> v;
            range_add(l, r, v);
        } else if (op == 2) {
            int l, r;
            long long x;
            cin >> l >> r >> x;
            cout << query_less(l, r, x) << "\n";
        } else {
            int l, r;
            long long x;
            cin >> l >> r >> x;
            cout << query_pred(l, r, x) << "\n";
        }
    }

    return 0;
}

    复杂度：O(N√N log √N)，块内排序 O(B log B)，查询 O(log B)。

    7. 整数分块/数论分块¶

    问题¶

    计算 \(\sum_{i=1}^{N} \lfloor N/i \rfloor\)

    核心思想¶

    \(\lfloor N/i \rfloor\) 的取值只有 O(√N) 种。

    • 当 \(i \leq \sqrt{N}\) 时，最多 √N 种取值
    • 当 \(i > \sqrt{N}\) 时，\(\lfloor N/i \rfloor \leq \sqrt{N}\)，最多 √N 种取值

    对于相同的 \(\lfloor N/i \rfloor = k\)，找到最大的 \(j\) 使得 \(\lfloor N/j \rfloor = k\)，则 \(j = \lfloor N/k \rfloor\)。

    #include <bits/stdc++.h>
using namespace std;

int main() {
    long long n;
    cin >> n;

    long long ans = 0;
    for (long long l = 1, r; l <= n; l = r + 1) {
        long long k = n / l;
        r = n / k;  // 最大的 j 使得 n/j = k
        ans += k * (r - l + 1);
    }

    cout << ans << "\n";
    return 0;

    复杂度：O(√N)

    应用：计算 \(\sum_{i=1}^{N} K \bmod i\)¶

    \(K \bmod i = K - i \cdot \lfloor K/i \rfloor\)

    \(\sum_{i=1}^{N} K \bmod i = NK - \sum_{i=1}^{N} i \cdot \lfloor K/i \rfloor\)

    对于相同的 \(\lfloor K/i \rfloor = k\)，\(i\) 的范围是 \([l, r]\)，贡献为 \(k \cdot \sum_{i=l}^{r} i = k \cdot \frac{(l+r)(r-l+1)}{2}\)

    for (long long l = 1, r; l <= n; l = r + 1) {
        long long q = k / l;
        if (q == 0) break;
        r = min(n, k / q);
        long long sum = (l + r) * (r - l + 1) / 2;
        ans -= q * sum;
    }

    cout << ans << "\n";

}

                p = p + a[p] + k;
                ans++;
            }
            cout << ans << "\n";
        }
    }

const int MAXN = 200005;
vector<int> blk[1005];  // 每个块

int tot;  // 总元素数

void init() {
    B = sqrt(n);
    blk[1].clear();
    tot = 0;

// 重新分块
void rebuild() {
    vector<int> all;
    for (int i = 1; i <= 1000; i++) {
        for (int x : blk[i]) {
            all.push_back(x);
        }
        blk[i].clear();
    }

    int sz = all.size();
    B = sqrt(sz);
    int num = (sz + B - 1) / B;

    for (int i = 0; i < sz; i++) {
        blk[i / B + 1].push_back(all[i]);
    }
}

// 在第 pos 个位置前插入 x
void insert(int pos, int x) {
    for (int i = 1; i <= 1000; i++) {
        if (pos <= blk[i].size()) {
            blk[i].insert(blk[i].begin() + pos - 1, x);
            tot++;
            if (blk[i].size() > 2 * B) {
                rebuild();
            }
            return;
        }
        pos -= blk[i].size();
    }
}

// 查询第 pos 个元素
int query(int pos) {
    for (int i = 1; i <= 1000; i++) {
        if (pos <= blk[i].size()) {
            return blk[i][pos - 1];
        }
        pos -= blk[i].size();
    }
    return -1;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> n >> m;
    init();

    for (int i = 1; i <= n; i++) {
        int x;
        cin >> x;
        insert(i, x);
    }

    while (m--) {
        int op;
        cin >> op;
        if (op == 1) {
            int pos, x;
            cin >> pos >> x;
            insert(pos, x);
        } else {
            int pos;
            cin >> pos;
            cout << query(pos) << "\n";
        }
    }

    return 0;
}

    10. 树上距离查询（CF348E）¶

    问题¶

    给定一棵树，支持： 1. 把节点 v 标记为红点 2. 查询节点 v 到最近红点的距离

    核心思想¶

    两种暴力： 1. 查询时暴力枚举所有红点，O(M) 每次查询 2. 修改时 BFS 更新所有节点到最近红点的距离，O(N) 每次修改

    总复杂度 O(N√M + M√M)。

    #include <bits/stdc++.h>
using namespace std;

const int MAXN = 100005;
vector<int> adj[MAXN];
int dist[MAXN];  // 到最近红点的距离
int ans_dist[MAXN];
vector<int> red_points;  // 当前块内的红点
int n, m;
int B;

void bfs(const vector<int>& sources) {
    queue<int> q;
    for (int i = 1; i <= n; i++) dist[i] = 1e9;
    for (int s : sources) {
        dist[s] = 0;
        q.push(s);
    }
    while (!q.empty()) {
        int u = q.front();
        q.pop();
        for (int v : adj[u]) {
            if (dist[v] > dist[u] + 1) {
                dist[v] = dist[u] + 1;
                q.push(v);
            }
        }
    }
}

int get_dist(int u, int v) {
    // LCA 求距离，这里简化为 BFS
    // 实际实现需要用倍增或树链剖分
    return abs(u - v);  // 占位
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> n >> m;
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }

    B = sqrt(m);

    vector<pair<int, int>> ops(m);
    for (int i = 0; i < m; i++) {
        cin >> ops[i].first >> ops[i].second;
    }

    for (int i = 0; i < m; i += B) {
        int end = min(i + B, m);

        // 多元 BFS 预处理
        bfs(red_points);
        for (int j = 1; j <= n; j++) {
            ans_dist[j] = dist[j];
        }

        // 处理块内操作
        for (int j = i; j < end; j++) {
            if (ops[j].first == 1) {
                red_points.push_back(ops[j].second);
            } else {
                int ans = ans_dist[ops[j].second];
                for (int rp : red_points) {
                    ans = min(ans, get_dist(ops[j].second, rp));
                }
                cout << ans << "\n";
            }
        }
    }

    return 0;
}

    #include <bits/stdc++.h>
using namespace std;

const int MAXN = 100005;
int a[MAXN];
vector<int> pos[MAXN];  // 每个数出现的位置
int f[505][505];        // f[i][j] = 第 i 块到第 j 块的众数
int cnt[MAXN];
int bl[MAXN];
int L[MAXN], R[MAXN];
int n, m, B;
int num_blk;

void init() {
    B = 300;  // 手动调参
    num_blk = (n + B - 1) / B;

    for (int i = 1; i <= n; i++) {
        bl[i] = (i - 1) / B + 1;
        pos[a[i]].push_back(i);
    }

    // 预处理 f[i][j]
    for (int i = 1; i <= num_blk; i++) {
        memset(cnt, 0, sizeof(cnt));
        int mode = 0, max_cnt = 0;
        for (int j = i; j <= num_blk; j++) {
            for (int k = L[j]; k <= R[j]; k++) {
                cnt[a[k]]++;
                if (cnt[a[k]] > max_cnt || (cnt[a[k]] == max_cnt && a[k] < mode)) {
                    max_cnt = cnt[a[k]];
                    mode = a[k];
                }
            }
            f[i][j] = mode;
        }
    }
}

// 查询 x 在 [l, r] 中的出现次数
int count_in_range(int x, int l, int r) {
    auto it1 = lower_bound(pos[x].begin(), pos[x].end(), l);
    auto it2 = upper_bound(pos[x].begin(), pos[x].end(), r);
    return it2 - it1;
}

int query_mode(int l, int r) {
    int bl_l = bl[l], bl_r = bl[r];

    if (bl_l == bl_r || bl_l + 1 == bl_r) {
        // 暴力
        map<int, int> cnt;
        int mode = 0, max_cnt = 0;
        for (int i = l; i <= r; i++) {
            cnt[a[i]]++;
            if (cnt[a[i]] > max_cnt || (cnt[a[i]] == max_cnt && a[i] < mode)) {
                max_cnt = cnt[a[i]];
                mode = a[i];
            }
        }
        return mode;
    }

    int mode = f[bl_l + 1][bl_r - 1];
    int max_cnt = count_in_range(mode, l, r);

    // 检查左散块
    for (int i = l; i <= R[bl_l]; i++) {
        int c = count_in_range(a[i], l, r);
        if (c > max_cnt || (c == max_cnt && a[i] < mode)) {
            max_cnt = c;
            mode = a[i];
        }
    }

    // 检查右散块
    for (int i = L[bl_r]; i <= r; i++) {
        int c = count_in_range(a[i], l, r);
        if (c > max_cnt || (c == max_cnt && a[i] < mode)) {
            max_cnt = c;
            mode = a[i];
        }
    }

    return mode;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
    }

    for (int i = 1; i <= num_blk; i++) {
        L[i] = (i - 1) * B + 1;
        R[i] = min(i * B, n);
    }

    init();

    while (m--) {
        int l, r;
        cin >> l >> r;
        cout << query_mode(l, r) << "\n";
    }

    return 0;
}

    复杂度：预处理 O(N²/B)，查询 O(√N log N)。块大小需要手动调参。

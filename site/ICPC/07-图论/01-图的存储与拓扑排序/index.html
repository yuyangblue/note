    01 图的存储与拓扑排序¶

    图论是ICPC的半壁江山。一切图算法的前提是：把图存进计算机。三种存储方式各有适用场景，拓扑排序是DAG（有向无环图）的基础操作。

    1. 图的基本概念¶

    图 G = (V, E)，V 是顶点集合，E 是边集合。

    概念 含义
    有向图 边有方向，u→v 和 v→u 是两条不同的边
    无向图 边无方向，u-v 等价于 v-u
    带权图 每条边有一个权值（距离、费用等）
    度数 无向图中与顶点相连的边数；有向图分入度和出度
    路径 顶点序列 v₁→v₂→...→vₖ，相邻顶点间有边
    连通图 无向图中任意两点之间都有路径
    DAG 有向无环图（Directed Acyclic Graph）

    顶点编号： 竞赛中通常用 1~n 或 0~n-1，统一即可。

    2. 三种存储方式¶

    2.1 邻接矩阵¶

    思路： 用一个 n×n 的矩阵 adj[u][v] 表示 u 到 v 是否有边（或边权）。

    1
2
3
4
5
         1  2  3  4
  1  0  1  0  0
  2  0  0  1  1
  3  0  0  0  0
  4  0  0  1  0
    优点 缺点
    查询 O(1)，实现简单 空间 O(n²)，稀疏图浪费严重
    适合稠密图 无法存重边（多条u→v的边）

    2.2 邻接表¶

    思路： 每个顶点 u 维护一个列表，存所有从 u 出发的边。

    1
2
3
4
    1: [2]
2: [3, 4]
3: []
4: [3]
    优点 缺点
    空间 O(n+m)，适合稀疏图 查询 u→v 是否有边需要遍历，O(度数)
    可以存重边 实现稍复杂

    竞赛中最常用。用 vector<vector<int>> 或 vector<vector<Edge>> 实现。

    2.3 链式前向星（前向星）¶

    思路： 用数组模拟邻接表，比 vector 更快（连续内存，缓存友好）。

    三个数组： - head[u]：顶点 u 的第一条边在 edge 数组中的下标（初始为 -1） - edge[i].to：第 i 条边的终点 - edge[i].next：与第 i 条边同起点的下一条边的下标

    插入边 u→v： 

    1
2
    edge[++cnt] = {v, head[u]};  // 新边的next指向原来的第一条
head[u] = cnt;                 // u的第一条边更新为新边

    遍历 u 的所有边： 

    1
2
3
    for (int i = head[u]; i != -1; i = edge[i].next) {
    int v = edge[i].to;
}

    #include <bits/stdc++.h>
using namespace std;

const int MAXN = 1005;
const int MAXM = 100005;

// ========== 1. 邻接矩阵 ==========
int adj_matrix[MAXN][MAXN];  // adj[u][v] = 边权，0表示无边

void add_edge_matrix(int u, int v, int w = 1) {
    adj_matrix[u][v] = w;
    // 无向图: adj_matrix[v][u] = w;
}

// ========== 2. 邻接表（vector） ==========
struct Edge {
    int to, w;
};
vector<Edge> adj_list[MAXN];

void add_edge_list(int u, int v, int w = 1) {
    adj_list[u].push_back({v, w});
    // 无向图: adj_list[v].push_back({u, w});
}

// ========== 3. 链式前向星 ==========
struct ForwardEdge {
    int to, w, next;
};
ForwardEdge fedge[MAXM];
int fhead[MAXN];
int fcnt;

void init_forward() {
    memset(fhead, -1, sizeof(fhead));
    fcnt = 0;
}

void add_edge_forward(int u, int v, int w = 1) {
    fedge[++fcnt] = {v, w, fhead[u]};
    fhead[u] = fcnt;
    // 无向图再加一条反向边
}

int main() {
    int n = 4, m = 4;
    // 边: 1->2, 2->3, 2->4, 4->3

    // 邻接矩阵
    add_edge_matrix(1, 2);
    add_edge_matrix(2, 3);
    add_edge_matrix(2, 4);
    add_edge_matrix(4, 3);
    cout << "邻接矩阵 2->3: " << adj_matrix[2][3] << endl;  // 预期: 1
    cout << "邻接矩阵 1->3: " << adj_matrix[1][3] << endl;  // 预期: 0

    // 邻接表
    add_edge_list(1, 2);
    add_edge_list(2, 3);
    add_edge_list(2, 4);
    add_edge_list(4, 3);
    cout << "\n邻接表遍历:" << endl;
    for (int u = 1; u <= n; u++) {
        cout << u << ": ";
        for (auto& e : adj_list[u]) {
            cout << e.to << " ";
        }
        cout << endl;
    }

    // 链式前向星
    init_forward();
    add_edge_forward(1, 2);
    add_edge_forward(2, 3);
    add_edge_forward(2, 4);
    add_edge_forward(4, 3);
    cout << "\n链式前向星遍历:" << endl;
    for (int u = 1; u <= n; u++) {
        cout << u << ": ";
        for (int i = fhead[u]; i != -1; i = fedge[i].next) {
            cout << fedge[i].to << " ";
        }
        cout << endl;
    }

    return 0;
}

    选择建议： 稠密图（m 接近 n²）用邻接矩阵；稀疏图用邻接表或链式前向星。竞赛中 vector 邻接表最常用（写起来简单），链式前向星在极端卡时常时更快。无向图要加两条边（u→v 和 v→u）。

    3. 图的遍历：DFS 与 BFS¶

    3.1 DFS（深度优先搜索）¶

    从起点出发，一条路走到黑，走不通就回溯。用递归或栈实现。

    用途： 连通性判断、找路径、拓扑排序（DFS版）、树的遍历。

    3.2 BFS（广度优先搜索）¶

    从起点出发，一层一层往外扩展。用队列实现。

    用途： 求无权图最短路、层次遍历、连通块大小。

    #include <bits/stdc++.h>
using namespace std;

const int MAXN = 1005;
vector<int> adj[MAXN];
bool vis[MAXN];

// DFS：递归版
void dfs(int u) {
    vis[u] = true;
    cout << u << " ";  // 访问顶点u
    for (int v : adj[u]) {
        if (!vis[v]) {
            dfs(v);
        }
    }
}

// DFS：迭代版（用栈，避免递归栈溢出）
void dfs_iter(int start) {
    stack<int> st;
    st.push(start);
    vis[start] = true;
    while (!st.empty()) {
        int u = st.top();
        st.pop();
        cout << u << " ";
        // 注意：栈是后进先出，倒序遍历保证顺序和递归版一致
        for (auto it = adj[u].rbegin(); it != adj[u].rend(); ++it) {
            int v = *it;
            if (!vis[v]) {
                vis[v] = true;
                st.push(v);
            }
        }
    }
}

// BFS：用队列
void bfs(int start) {
    queue<int> q;
    q.push(start);
    vis[start] = true;
    while (!q.empty()) {
        int u = q.front();
        q.pop();
        cout << u << " ";
        for (int v : adj[u]) {
            if (!vis[v]) {
                vis[v] = true;
                q.push(v);
            }
        }
    }
}

// BFS求无权图最短路（距离数组）
vector<int> bfs_distance(int start, int n) {
    vector<int> dist(n + 1, -1);  // -1表示不可达
    queue<int> q;
    q.push(start);
    dist[start] = 0;
    while (!q.empty()) {
        int u = q.front();
        q.pop();
        for (int v : adj[u]) {
            if (dist[v] == -1) {
                dist[v] = dist[u] + 1;
                q.push(v);
            }
        }
    }
    return dist;
}

int main() {
    // 建图：1-2, 1-3, 2-4, 3-4, 4-5（无向图）
    adj[1].push_back(2); adj[2].push_back(1);
    adj[1].push_back(3); adj[3].push_back(1);
    adj[2].push_back(4); adj[4].push_back(2);
    adj[3].push_back(4); adj[4].push_back(3);
    adj[4].push_back(5); adj[5].push_back(4);

    cout << "DFS(递归) 从1出发: ";
    memset(vis, false, sizeof(vis));
    dfs(1);
    cout << endl;

    cout << "BFS 从1出发: ";
    memset(vis, false, sizeof(vis));
    bfs(1);
    cout << endl;

    cout << "BFS求距离(从1出发): ";
    vector<int> dist = bfs_distance(1, 5);
    for (int i = 1; i <= 5; i++) {
        cout << i << "(" << dist[i] << ") ";
    }
    cout << endl;
    // 预期: 1(0) 2(1) 3(1) 4(2) 5(3)

    return 0;
}

    坑： DFS递归在图很大（n>1e5）时可能栈溢出，用迭代版或手动加栈。BFS的 vis 标记要在入队时就打，不能在出队时打（否则同一个节点可能多次入队）。求距离用 dist == -1 判断未访问，比单独的 vis 数组更省空间。

    4. 拓扑排序¶

    大白话引入¶

    选课有先修关系：必须先学"高等数学"才能学"线性代数"，先学"C语言"才能学"数据结构"。把所有课程排成一个序列，使得每门课的先修课都排在它前面——这就是拓扑排序。

    前提：图必须是 DAG（有向无环图）。有环的话（A需要B先修，B需要A先修）不可能排序。

    4.1 Kahn 算法（BFS 版，最常用）¶

    思路： 1. 计算所有顶点的入度 2. 把入度为 0 的顶点加入队列 3. 取出队首 u，加入拓扑序列 4. 遍历 u 的所有出边 u→v，v 的入度减 1；如果 v 入度变为 0，加入队列 5. 重复直到队列为空 6. 如果拓扑序列长度 < n，说明有环（不存在拓扑排序）

    时间复杂度：O(n+m)

    4.2 DFS 版¶

    思路： DFS 遍历，顶点处理完所有邻居后加入栈中，最后逆序输出就是拓扑序列。

    #include <bits/stdc++.h>
using namespace std;

const int MAXN = 1005;
vector<int> adj[MAXN];
int in_degree[MAXN];  // 入度

// Kahn算法（BFS版）：返回拓扑序列，如果有环返回空vector
vector<int> topo_kahn(int n) {
    vector<int> result;
    queue<int> q;
    // 1. 所有入度为0的顶点入队
    for (int i = 1; i <= n; i++) {
        if (in_degree[i] == 0) {
            q.push(i);
        }
    }
    // 2. BFS
    while (!q.empty()) {
        int u = q.front();
        q.pop();
        result.push_back(u);
        for (int v : adj[u]) {
            in_degree[v]--;  // 入度减1
            if (in_degree[v] == 0) {
                q.push(v);
            }
        }
    }
    // 3. 判断是否有环
    if ((int)result.size() < n) {
        return {};  // 有环，返回空
    }
    return result;
}

// DFS版拓扑排序
bool dfs_vis[MAXN];  // 0=未访问, 1=访问中, 2=已完成
bool dfs_cycle;
vector<int> dfs_result;

void dfs_topo(int u) {
    dfs_vis[u] = true;  // 标记为访问中（用int数组更精确，这里简化）
    for (int v : adj[u]) {
        if (!dfs_vis[v]) {
            dfs_topo(v);
        }
    }
    dfs_result.push_back(u);  // 处理完所有邻居后加入
}

vector<int> topo_dfs(int n) {
    dfs_result.clear();
    memset(dfs_vis, false, sizeof(dfs_vis));
    for (int i = 1; i <= n; i++) {
        if (!dfs_vis[i]) {
            dfs_topo(i);
        }
    }
    reverse(dfs_result.begin(), dfs_result.end());  // 逆序输出
    return dfs_result;
}

int main() {
    int n = 6;
    // 边: 1->2, 1->3, 2->4, 3->4, 4->5, 4->6（DAG）
    int edges[][2] = {{1,2},{1,3},{2,4},{3,4},{4,5},{4,6}};
    for (auto& e : edges) {
        adj[e[0]].push_back(e[1]);
        in_degree[e[1]]++;
    }

    // Kahn算法
    vector<int> topo = topo_kahn(n);
    if (topo.empty()) {
        cout << "有环，不存在拓扑排序" << endl;
    } else {
        cout << "Kahn拓扑排序: ";
        for (int x : topo) cout << x << " ";
        cout << endl;
    }
    // 预期（一种可能）: 1 2 3 4 5 6 或 1 3 2 4 6 5 等

    // DFS版（注意：in_degree已被Kahn修改，重新建图）
    // 这里简化，直接用DFS版
    cout << "DFS拓扑排序: ";
    vector<int> topo2 = topo_dfs(n);
    for (int x : topo2) cout << x << " ";
    cout << endl;

    // 测试有环的情况
    cout << "\n测试有环: 3->1 加入后" << endl;
    // 重新初始化
    for (int i = 1; i <= n; i++) {
        adj[i].clear();
        in_degree[i] = 0;
    }
    for (auto& e : edges) {
        adj[e[0]].push_back(e[1]);
        in_degree[e[1]]++;
    }
    adj[3].push_back(1);  // 加一条 3->1，形成环 1->3->1
    in_degree[1]++;
    vector<int> topo3 = topo_kahn(n);
    if (topo3.empty()) {
        cout << "检测到环，不存在拓扑排序" << endl;
    } else {
        cout << "拓扑排序: ";
        for (int x : topo3) cout << x << " ";
        cout << endl;
    }

    return 0;
}

    坑： Kahn算法中入度数组会被修改，多次调用要重新计算。判断有环的标准是 result.size() < n（不是队列空了就一定有环，而是看处理了多少个顶点）。DFS版需要用三色标记（0未访问/1访问中/2已完成）才能准确判环，简化版可能误判。拓扑排序不唯一——只要满足先修关系都是合法的。

    洛谷题¶

    题号 题目 难度 提示
    P3916 图的遍历 ⭐ 普及- 反向建图+DFS，从大到小遍历求每个点能到达的最大编号
    B3643 图的存储与遍历 ⭐ 入门 纯模板：邻接矩阵/邻接表存储，DFS/BFS遍历
    P1113 杂务 ⭐ 普及- 拓扑排序+DP：求完成所有杂务的最短时间，拓扑序上做DP
    P4017 最大食物链计数 ⭐ 普及- 拓扑排序+DP：DAG上计数，入度0的是起点，出度0的是终点
    P1038 [NOIP2003 提高组] 神经网络 ⭐⭐ 普及+/提高 拓扑排序：按拓扑序传递神经元状态，输出层输出>0的神经元
    P1983 [NOIP2013 普及组] 车站分级 ⭐⭐ 普及+/提高 拓扑排序+差分：根据车次停靠关系建图，求最长拓扑层数
    P1347 排序 ⭐⭐ 普及+/提高 拓扑排序：每加一条边做一次拓扑，判断是否有唯一序/有环/不确定

    本节重点： - 三种存储：邻接矩阵（稠密图O(n²)）、邻接表（稀疏图O(n+m)，最常用）、链式前向星（最快） - 无向图加两条边，有向图加一条 - DFS递归/迭代，BFS用队列，入队时打vis标记 - 拓扑排序前提是DAG，Kahn算法用入度+队列，O(n+m) - 拓扑排序不唯一，result.size()<n 说明有环 - 常见坑：无向图忘加反向边；BFS出队才打vis导致重复入队；Kahn多次调用忘重置入度；DFS递归n太大会栈溢出

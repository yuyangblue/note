    01 字符串哈希¶

    字符串哈希把字符串映射成一个数字，让两个字符串的比较从 O(n) 变成 O(1)。是字符串题中最常用的工具之一。

    1. 核心思想¶

    大白话引入¶

    比较两个字符串是否相等，暴力做法是逐字符比较，O(n)。如果要比较很多次（比如 10⁵ 次），就太慢了。

    字符串哈希：把字符串看成一个"base进制的数"，算出它的哈希值。两个字符串相等 → 哈希值相等（大概率）。比较哈希值只需 O(1)。

    比如字符串 "abc"，base=131： hash = a·base² + b·base¹ + c·base⁰ = 97·131² + 98·131 + 99

    核心讲解¶

    多项式哈希公式： $\(hash(s) = s_0 \cdot base^{n-1} + s_1 \cdot base^{n-2} + \dots + s_{n-1} \cdot base^0\)$

    其中 sᵢ 是第 i 个字符的 ASCII 值（或映射到 1~26，避免 0 带来的问题）。

    base 的选择： 通常用 131 或 13331（经验值，冲突少）。base 要大于字符集大小。

    模数的选择： - 自然溢出：用 unsigned long long，溢出时自动 mod 2⁶⁴，速度快但有被卡的风险 - 取大质数：如 1e9+7、998244353，更安全 - 双哈希：用两个不同的 base 和 mod，冲突概率极低

    为什么字符要映射到 1~26 而不是 0~25？ 如果用 0，"a" 和 "aa" 的前缀哈希可能产生歧义（0 和 0·base+0 都是0）。用 1 开始避免前导零问题。竞赛中直接用 ASCII 值（从 97 开始）也可以。

    2. 前缀哈希与子串查询¶

    思维启发¶

    如果每次都从头算哈希，还是 O(n)。用前缀和的思想预处理，就能 O(1) 查询任意子串的哈希。

    定义前缀哈希 h[i] = hash(s[0..i-1])（前 i 个字符的哈希）。

    则子串 s[l..r]（0-indexed，包含 l 和 r）的哈希： $\(hash(s[l..r]) = h[r+1] - h[l] \cdot base^{r-l+1}\)$

    推导： h[r+1] = s₀·base^r + ... + s_l·base^(r-l) + ... + s_r h[l]·base^(r-l+1) = s₀·base^r + ... + s_{l-1}·base^(r-l+1) 相减后只剩下 s_l·base^(r-l) + ... + s_r，正好是子串的哈希。

    #include <bits/stdc++.h>
using namespace std;

typedef unsigned long long ull;

const int MAXN = 1005;
const ull BASE = 131;

ull h[MAXN];       // 前缀哈希
ull pw[MAXN];      // base 的幂次

// 预处理前缀哈希和幂次
void build_hash(const string& s) {
    int n = s.size();
    pw[0] = 1;
    for (int i = 1; i <= n; i++) {
        pw[i] = pw[i - 1] * BASE;
    }
    h[0] = 0;
    for (int i = 0; i < n; i++) {
        // h[i+1] = h[i] * base + s[i]
        h[i + 1] = h[i] * BASE + s[i];
    }
}

// 查询子串 s[l..r] 的哈希值（0-indexed，闭区间）
ull get_hash(int l, int r) {
    return h[r + 1] - h[l] * pw[r - l + 1];
}

int main() {
    string s = "abcabc";
    build_hash(s);

    // 测试："abc" 出现两次，哈希值应该相同
    cout << "s[0..2] (abc) hash: " << get_hash(0, 2) << endl;
    cout << "s[3..5] (abc) hash: " << get_hash(3, 5) << endl;
    cout << "相同? " << (get_hash(0, 2) == get_hash(3, 5) ? "是" : "否") << endl;

    // 不同子串哈希不同
    cout << "s[0..0] (a) hash: " << get_hash(0, 0) << endl;
    cout << "s[1..1] (b) hash: " << get_hash(1, 1) << endl;

    // 整串哈希
    cout << "整串 hash: " << get_hash(0, 5) << endl;

    return 0;
}

    坑： 子串哈希公式中的 h[l] * pw[r-l+1] 不能写反——是 h[l] 乘以幂次，不是 h[r+1]。区间是 0-indexed 闭区间 [l,r]，长度是 r-l+1。自然溢出用 unsigned long long，减法自动 mod 2⁶⁴。

    3. 单哈希实现¶

    自然溢出 vs 取模¶

    方式 类型 优点 缺点
    自然溢出 unsigned long long 速度快，不用手动取模 有被 Hack 风险（构造冲突串）
    取大质数 long long mod 1e9+7 较安全 每步要手动取模，稍慢

    竞赛中：一般题自然溢出够用；被卡过的题（或CF等Hack机制）用双哈希。

    #include <bits/stdc++.h>
using namespace std;

const int MAXN = 1005;
const long long MOD = 1e9 + 7;
const long long BASE = 131;

long long h[MAXN];
long long pw[MAXN];

void build_hash(const string& s) {
    int n = s.size();
    pw[0] = 1;
    for (int i = 1; i <= n; i++) {
        pw[i] = pw[i - 1] * BASE % MOD;
    }
    h[0] = 0;
    for (int i = 0; i < n; i++) {
        h[i + 1] = (h[i] * BASE + s[i]) % MOD;
    }
}

long long get_hash(int l, int r) {
    // 减法可能为负，加 MOD 再取模
    return (h[r + 1] - h[l] * pw[r - l + 1] % MOD + MOD) % MOD;
}

int main() {
    string s = "hello world";
    build_hash(s);
    cout << "整串哈希: " << get_hash(0, s.size() - 1) << endl;
    cout << "hello 哈希: " << get_hash(0, 4) << endl;
    cout << "world 哈希: " << get_hash(6, 10) << endl;
    return 0;
}

    4. 双哈希（防冲突）¶

    思维启发¶

    单哈希有极小概率冲突（两个不同字符串哈希值相同）。用两组不同的 base 和 mod 同时算哈希，比较时两个都要相等。冲突概率从 1/p 降到 1/(p₁·p₂)， practically 为零。

    #include <bits/stdc++.h>
using namespace std;

const int MAXN = 1005;

// 第一组哈希
const long long MOD1 = 1e9 + 7;
const long long BASE1 = 131;
long long h1[MAXN], pw1[MAXN];

// 第二组哈希
const long long MOD2 = 998244353;
const long long BASE2 = 13331;
long long h2[MAXN], pw2[MAXN];

// 用 pair 表示双哈希值
typedef pair<long long, long long> pll;

void build_hash(const string& s) {
    int n = s.size();
    pw1[0] = pw2[0] = 1;
    for (int i = 1; i <= n; i++) {
        pw1[i] = pw1[i - 1] * BASE1 % MOD1;
        pw2[i] = pw2[i - 1] * BASE2 % MOD2;
    }
    h1[0] = h2[0] = 0;
    for (int i = 0; i < n; i++) {
        h1[i + 1] = (h1[i] * BASE1 + s[i]) % MOD1;
        h2[i + 1] = (h2[i] * BASE2 + s[i]) % MOD2;
    }
}

pll get_hash(int l, int r) {
    long long v1 = (h1[r + 1] - h1[l] * pw1[r - l + 1] % MOD1 + MOD1) % MOD1;
    long long v2 = (h2[r + 1] - h2[l] * pw2[r - l + 1] % MOD2 + MOD2) % MOD2;
    return {v1, v2};
}

int main() {
    string s = "abababab";
    build_hash(s);

    // "ab" 出现多次，双哈希都应相同
    pll h_ab1 = get_hash(0, 1);
    pll h_ab2 = get_hash(2, 3);
    pll h_ab3 = get_hash(4, 5);
    cout << "ab 双哈希1: (" << h_ab1.first << ", " << h_ab1.second << ")" << endl;
    cout << "ab 双哈希2: (" << h_ab2.first << ", " << h_ab2.second << ")" << endl;
    cout << "全部相同? " << (h_ab1 == h_ab2 && h_ab2 == h_ab3 ? "是" : "否") << endl;

    return 0;
}

    坑： 双哈希的两组 base 和 mod 必须不同。比较时用 pair<ll,ll> 的 == 运算符。数组要开两倍（h1/h2, pw1/pw2）。双哈希代码量大，可以封装成 struct 或 class 减少重复。

    5. 常见应用¶

    5.1 字符串匹配（O(n+m)）¶

    文本串 T 长度 n，模式串 P 长度 m。算 P 的哈希，然后滑动窗口算 T 中所有长度为 m 的子串哈希，比较即可。

    比 KMP 简单，但常数稍大。

    5.2 最长回文子串（二分+哈希，O(n log n)）¶

    二分回文半径，用哈希判断正串和反串是否相等。需要同时预处理正串和反串的哈希。

    5.3 最长公共前缀（二分+哈希，O(log n) 每次查询）¶

    两个后缀的 LCP，二分长度，用哈希判断子串是否相等。

    5.4 字符串去重¶

    把所有字符串的哈希值存入 set 或排序，O(n log n) 去重。

    #include <bits/stdc++.h>
using namespace std;

typedef unsigned long long ull;
const ull BASE = 131;

// 自然溢出版字符串匹配
// 返回模式串 P 在文本串 T 中所有出现位置（0-indexed）
vector<int> string_match(const string& T, const string& P) {
    vector<int> positions;
    int n = T.size(), m = P.size();
    if (m > n) return positions;

    // 算模式串哈希
    ull hash_p = 0;
    for (char c : P) hash_p = hash_p * BASE + c;

    // 算文本串前缀哈希
    vector<ull> h(n + 1, 0), pw(n + 1, 1);
    for (int i = 1; i <= n; i++) {
        pw[i] = pw[i - 1] * BASE;
        h[i] = h[i - 1] * BASE + T[i - 1];
    }

    // 滑动窗口比较
    for (int i = 0; i + m <= n; i++) {
        ull hash_sub = h[i + m] - h[i] * pw[m];
        if (hash_sub == hash_p) {
            positions.push_back(i);
        }
    }
    return positions;
}

int main() {
    string T = "ababcabab";
    string P = "ab";
    vector<int> pos = string_match(T, P);
    cout << "\"" << P << "\" 在 \"" << T << "\" 中出现位置: ";
    for (int p : pos) cout << p << " ";
    cout << endl;
    // 预期: 0 2 5 7

    P = "abc";
    pos = string_match(T, P);
    cout << "\"" << P << "\" 出现位置: ";
    for (int p : pos) cout << p << " ";
    cout << endl;
    // 预期: 2

    return 0;
}

    1. 大白话引入¶

    想象你有一本 1000 页的书，老师让你快速找到第 300-700 页里有多少个"算法"这个词。

    暴力做法：一页一页翻，从第 300 页翻到第 700 页。太慢了。

    分块思想：把书分成 30 块，每块约 33 页。提前统计好每块里有多少个"算法"。查询时： - 中间的整块（第 10-21 块）：直接查预处理的答案，O(1) - 两端的散块（第 9 块的一部分、第 22 块的一部分）：暴力数

    这样查询复杂度从 O(区间长度) 降到 O(块数 + 块大小) = O(√N)。

    分块 vs 线段树： - 线段树：递归结构，log N，但需要信息可合并 - 分块：两层结构（整块 + 散块），√N，但不需要合并，能维护更复杂的信息

    2. 基础分块：区间加、单点查询、区间和¶

    核心思想¶

    把长度为 N 的序列分成 √N 个块，每块大小 B = √N。

    区间加操作： - 整块：打懒标记 add[i]，表示第 i 块所有数都要加上这个值，O(1) - 散块：暴力修改原数组，O(B)

    单点查询： - 原数组值 + 所在块的懒标记

    区间和查询： - 整块：预处理块的和 sum[i]，查询时 sum[i] + add[i] * B - 散块：暴力累加 a[j] + add[块]

    #include <bits/stdc++.h>
using namespace std;

const int MAXN = 100005;
int a[MAXN];           // 原数组
long long sum[MAXN];   // 每块的和
long long add[MAXN];   // 每块的懒标记
int bl[MAXN];          // bl[i] 表示第 i 个元素属于第几块
int L[MAXN], R[MAXN];  // 每块的左右端点
int n, m, B;           // B 是块大小

// 初始化分块
void init() {
    B = sqrt(n);
    int num = (n + B - 1) / B;  // 块数
    for (int i = 1; i <= num; i++) {
        L[i] = (i - 1) * B + 1;
        R[i] = min(i * B, n);
    }
    for (int i = 1; i <= n; i++) {
        bl[i] = (i - 1) / B + 1;
        sum[bl[i]] += a[i];
    }
}

// 区间加：给 [l, r] 加上 v
void range_add(int l, int r, int v) {
    int bl_l = bl[l], bl_r = bl[r];

    // 如果在同一块，暴力修改
    if (bl_l == bl_r) {
        for (int i = l; i <= r; i++) {
            a[i] += v;
            sum[bl_l] += v;
        }
        return;
    }

    // 左散块：暴力修改
    for (int i = l; i <= R[bl_l]; i++) {
        a[i] += v;
        sum[bl_l] += v;
    }

    // 中间整块：打懒标记
    for (int i = bl_l + 1; i < bl_r; i++) {
        add[i] += v;
    }

    // 右散块：暴力修改
    for (int i = L[bl_r]; i <= r; i++) {
        a[i] += v;
        sum[bl_r] += v;
    }
}

// 单点查询：查询位置 pos 的值
long long query_point(int pos) {
    return a[pos] + add[bl[pos]];
}

// 区间和查询：查询 [l, r] 的和
long long query_sum(int l, int r) {
    int bl_l = bl[l], bl_r = bl[r];
    long long ans = 0;

    // 如果在同一块，暴力累加
    if (bl_l == bl_r) {
        for (int i = l; i <= r; i++) {
            ans += a[i] + add[bl_l];
        }
        return ans;
    }

    // 左散块
    for (int i = l; i <= R[bl_l]; i++) {
        ans += a[i] + add[bl_l];
    }

    // 中间整块
    for (int i = bl_l + 1; i < bl_r; i++) {
        ans += sum[i] + add[i] * (R[i] - L[i] + 1);
    }

    // 右散块
    for (int i = L[bl_r]; i <= r; i++) {
        ans += a[i] + add[bl_r];
    }

    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
    }

    init();

    while (m--) {
        int op, l, r;
        cin >> op >> l >> r;
        if (op == 1) {  // 区间加
            int v;
            cin >> v;
            range_add(l, r, v);
        } else {  // 区间和查询
            cout << query_sum(l, r) << "\n";
        }
    }

    return 0;
}

    复杂度分析： - 区间加：O(B + N/B)，当 B = √N 时最优，O(√N) - 单点查询：O(1) - 区间和查询：O(B + N/B) = O(√N)

    3. 进阶：区间乘与标记下放¶

    问题¶

    现在要支持两种操作： 1. 区间乘：给 [l, r] 所有数乘上 v 2. 区间加：给 [l, r] 所有数加上 v

    核心思想¶

    每个块维护两个标记：mul[i]（乘法标记）和 add[i]（加法标记）。

    块内元素的实际值 = a[i] * mul[块] + add[块]

    区间乘 v： - 整块：mul[i] *= v，add[i] *= v - 散块：先下放标记（把标记作用到原数组），再暴力修改

    区间加 v： - 整块：add[i] += v - 散块：先下放标记，再暴力修改

    标记下放：把 mul 和 add 作用到原数组，然后重置标记。

    // 区间乘+区间加的标记下放与操作（思路示意，完整实现见基础分块章节）
#include <bits/stdc++.h>
using namespace std;

const int MAXN = 100005;
long long a[MAXN];
long long sum[MAXN];
long long mul[MAXN], add[MAXN];
int bl[MAXN];
int L[MAXN], R[MAXN];
int n, B;

// 下放第 b 块的标记
void push_down(int b) {
    if (mul[b] != 1 || add[b] != 0) {
        for (int i = L[b]; i <= R[b]; i++) {
            a[i] = a[i] * mul[b] + add[b];
        }
        mul[b] = 1;
        add[b] = 0;
    }
}

// 区间乘
void range_mul(int l, int r, int v) {
    int bl_l = bl[l], bl_r = bl[r];

    if (bl_l == bl_r) {
        push_down(bl_l);
        for (int i = l; i <= r; i++) {
            a[i] *= v;
        }
        sum[bl_l] = 0;
        for (int i = L[bl_l]; i <= R[bl_l]; i++) {
            sum[bl_l] += a[i];
        }
        return;
    }

    push_down(bl_l);
    for (int i = l; i <= R[bl_l]; i++) {
        a[i] *= v;
    }
    sum[bl_l] = 0;
    for (int i = L[bl_l]; i <= R[bl_l]; i++) {
        sum[bl_l] += a[i];
    }

    for (int i = bl_l + 1; i < bl_r; i++) {
        mul[i] *= v;
        add[i] *= v;
        sum[i] *= v;
    }

    push_down(bl_r);
    for (int i = L[bl_r]; i <= r; i++) {
        a[i] *= v;
    }
    sum[bl_r] = 0;
    for (int i = L[bl_r]; i <= R[bl_r]; i++) {
        sum[bl_r] += a[i];
    }
}

// 区间加
void range_add(int l, int r, int v) {
    int bl_l = bl[l], bl_r = bl[r];

    if (bl_l == bl_r) {
        push_down(bl_l);
        for (int i = l; i <= r; i++) {
            a[i] += v;
        }
        sum[bl_l] = 0;
        for (int i = L[bl_l]; i <= R[bl_l]; i++) {
            sum[bl_l] += a[i];
        }
        return;
    }

    push_down(bl_l);
    for (int i = l; i <= R[bl_l]; i++) {
        a[i] += v;
    }
    sum[bl_l] = 0;
    for (int i = L[bl_l]; i <= R[bl_l]; i++) {
        sum[bl_l] += a[i];
    }

    for (int i = bl_l + 1; i < bl_r; i++) {
        add[i] += v;
        sum[i] += v * (R[i] - L[i] + 1);
    }

    push_down(bl_r);
    for (int i = L[bl_r]; i <= r; i++) {
        a[i] += v;
    }
    sum[bl_r] = 0;
    for (int i = L[bl_r]; i <= R[bl_r]; i++) {
        sum[bl_r] += a[i];
    }
}

int main() { return 0; }

    关键点：散块修改前必须下放标记，否则标记会覆盖之前的操作。

    4. 区间开方（均摊思想）¶

    问题¶

    支持两种操作： 1. 区间开方：把 [l, r] 所有数向下取整开方 2. 区间求和

    核心思想¶

    一个数最多开方 5 次就会变成 1（int 范围内，2^31 → 2^15 → 2^7 → 2^3 → 2^1 → 1）。

    均摊分析：每个块最多被暴力修改 5 次，之后块内全是 1，不需要再修改。

    维护： - cnt[i]：第 i 块被开方的次数 - 如果 cnt[i] >= 5，跳过这个块

    #include <bits/stdc++.h>
using namespace std;

const int MAXN = 100005;
long long a[MAXN];
long long sum[MAXN];
int cnt[MAXN];  // 开方次数
int bl[MAXN];
int L[MAXN], R[MAXN];
int n, m, B;

void init() {
    B = sqrt(n);
    int num = (n + B - 1) / B;
    for (int i = 1; i <= num; i++) {
        L[i] = (i - 1) * B + 1;
        R[i] = min(i * B, n);
    }
    for (int i = 1; i <= n; i++) {
        bl[i] = (i - 1) / B + 1;
        sum[bl[i]] += a[i];
    }
}

void range_sqrt(int l, int r) {
    int bl_l = bl[l], bl_r = bl[r];

    if (bl_l == bl_r) {
        for (int i = l; i <= r; i++) {
            sum[bl_l] -= a[i];
            a[i] = sqrt(a[i]);
            sum[bl_l] += a[i];
        }
        return;
    }

    // 左散块
    for (int i = l; i <= R[bl_l]; i++) {
        sum[bl_l] -= a[i];
        a[i] = sqrt(a[i]);
        sum[bl_l] += a[i];
    }

    // 中间整块
    for (int i = bl_l + 1; i < bl_r; i++) {
        if (cnt[i] >= 5) continue;  // 已经全是 1，跳过
        cnt[i]++;
        sum[i] = 0;
        for (int j = L[i]; j <= R[i]; j++) {
            a[j] = sqrt(a[j]);
            sum[i] += a[j];
        }
    }

    // 右散块
    for (int i = L[bl_r]; i <= r; i++) {
        sum[bl_r] -= a[i];
        a[i] = sqrt(a[i]);
        sum[bl_r] += a[i];
    }
}

long long query_sum(int l, int r) {
    int bl_l = bl[l], bl_r = bl[r];
    long long ans = 0;

    if (bl_l == bl_r) {
        for (int i = l; i <= r; i++) {
            ans += a[i];
        }
        return ans;
    }

    for (int i = l; i <= R[bl_l]; i++) {
        ans += a[i];
    }
    for (int i = bl_l + 1; i < bl_r; i++) {
        ans += sum[i];
    }
    for (int i = L[bl_r]; i <= r; i++) {
        ans += a[i];
    }

    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
    }

    init();

    while (m--) {
        int op, l, r;
        cin >> op >> l >> r;
        if (op == 1) {
            range_sqrt(l, r);
        } else {
            cout << query_sum(l, r) << "\n";
        }
    }

    return 0;
}

    复杂度：均摊 O(N√N)，因为每个块最多被暴力修改 5 次。

    5. 区间数颜色（均摊分析）¶

    问题¶

    支持两种操作： 1. 查询 [l, r] 中值为 c 的元素个数 2. 把 [l, r] 所有数改成 c

    核心思想¶

    均摊分析：每次区间覆盖会把若干个"颜色段"合并成一个。

    • 初始 N 个独立颜色段
    • 每次覆盖操作最多新增 2 个颜色段（左右两端）
    • 每次覆盖会消除若干个颜色段

    每个颜色段消失时贡献 O(长度) 的复杂度，总复杂度 O(N√N)。

    实现： - 每个块维护一个标记 tag[i]，表示整块被染成的颜色 - 如果 tag[i] != -1，整块都是这个颜色 - 散块修改时，先下放标记，再暴力修改

    #include <bits/stdc++.h>
using namespace std;

const int MAXN = 100005;
int a[MAXN];
int tag[MAXN];  // tag[i] = -1 表示没有统一颜色
int bl[MAXN];
int L[MAXN], R[MAXN];
int n, m, B;

void init() {
    B = sqrt(n);
    int num = (n + B - 1) / B;
    for (int i = 1; i <= num; i++) {
        L[i] = (i - 1) * B + 1;
        R[i] = min(i * B, n);
        tag[i] = -1;
    }
    for (int i = 1; i <= n; i++) {
        bl[i] = (i - 1) / B + 1;
    }
}

// 下放标记
void push_down(int b) {
    if (tag[b] != -1) {
        for (int i = L[b]; i <= R[b]; i++) {
            a[i] = tag[b];
        }
        tag[b] = -1;
    }
}

// 查询 [l, r] 中 c 的个数
int query_count(int l, int r, int c) {
    int bl_l = bl[l], bl_r = bl[r];
    int ans = 0;

    if (bl_l == bl_r) {
        push_down(bl_l);
        for (int i = l; i <= r; i++) {
            if (a[i] == c) ans++;
        }
        return ans;
    }

    push_down(bl_l);
    for (int i = l; i <= R[bl_l]; i++) {
        if (a[i] == c) ans++;
    }

    for (int i = bl_l + 1; i < bl_r; i++) {
        if (tag[i] == c) {
            ans += R[i] - L[i] + 1;
        } else if (tag[i] == -1) {
            for (int j = L[i]; j <= R[i]; j++) {
                if (a[j] == c) ans++;
            }
        }
    }

    push_down(bl_r);
    for (int i = L[bl_r]; i <= r; i++) {
        if (a[i] == c) ans++;
    }

    return ans;
}

// 把 [l, r] 改成 c
void range_assign(int l, int r, int c) {
    int bl_l = bl[l], bl_r = bl[r];

    if (bl_l == bl_r) {
        push_down(bl_l);
        for (int i = l; i <= r; i++) {
            a[i] = c;
        }
        return;

    push_down(bl_l);
    for (int i = l; i <= R[bl_l]; i++) {
        a[i] = c;
    }

    for (int i = bl_l + 1; i < bl_r; i++) {
        tag[i] = c;
    }

    push_down(bl_r);
    for (int i = L[bl_r]; i <= r; i++) {
        a[i] = c;
    }
}

using namespace std;

const int MAXN = 100005;
int a[MAXN];
vector<int> sorted_blk[MAXN / 300 + 5];  // 每块的排序副本
long long add[MAXN / 300 + 5];           // 懒标记
int bl[MAXN];
int L[MAXN], R[MAXN];
int n, m, B;
int num_blk;

void init() {
    B = sqrt(n);
    num_blk = (n + B - 1) / B;
    for (int i = 1; i <= num_blk; i++) {
        L[i] = (i - 1) * B + 1;
        R[i] = min(i * B, n);
        sorted_blk[i].clear();
        for (int j = L[i]; j <= R[i]; j++) {

        }

// 重建第 b 块的排序副本
void rebuild(int b) {
    sorted_blk[b].clear();
    for (int i = L[b]; i <= R[b]; i++) {
        sorted_blk[b].push_back(a[i] + add[b]);
    }
    sort(sorted_blk[b].begin(), sorted_blk[b].end());
}

// 区间加
void range_add(int l, int r, int v) {
    int bl_l = bl[l], bl_r = bl[r];

    if (bl_l == bl_r) {
        for (int i = l; i <= r; i++) {
            a[i] += v;
        }
        rebuild(bl_l);
        return;
    }

    for (int i = l; i <= R[bl_l]; i++) {
        a[i] += v;
    }
    rebuild(bl_l);

    for (int i = bl_l + 1; i < bl_r; i++) {
        add[i] += v;
    }

    for (int i = L[bl_r]; i <= r; i++) {
        a[i] += v;
    }
    rebuild(bl_r);
}

// 查询 [l, r] 中小于 x 的个数
int query_less(int l, int r, long long x) {
    int bl_l = bl[l], bl_r = bl[r];
    int ans = 0;

    if (bl_l == bl_r) {
        for (int i = l; i <= r; i++) {
            if (a[i] + add[bl_l] < x) ans++;
        }
        return ans;
    }

    for (int i = l; i <= R[bl_l]; i++) {
        if (a[i] + add[bl_l] < x) ans++;
    }

    for (int i = bl_l + 1; i < bl_r; i++) {
        // 在 sorted_blk[i] 中二分查找 x - add[i]
        int cnt = lower_bound(sorted_blk[i].begin(), sorted_blk[i].end(), x - add[i]) - sorted_blk[i].begin();
        ans += cnt;
    }

    for (int i = L[bl_r]; i <= r; i++) {
        if (a[i] + add[bl_r] < x) ans++;
    }

    return ans;
}

// 查询 [l, r] 中 x 的前驱
long long query_pred(int l, int r, long long x) {
    int bl_l = bl[l], bl_r = bl[r];
    long long ans = -1e18;

    if (bl_l == bl_r) {
        for (int i = l; i <= r; i++) {
            long long val = a[i] + add[bl_l];
            if (val < x) ans = max(ans, val);
        }
        return ans;
    }

    for (int i = l; i <= R[bl_l]; i++) {
        long long val = a[i] + add[bl_l];
        if (val < x) ans = max(ans, val);
    }

    for (int i = bl_l + 1; i < bl_r; i++) {
        // 在 sorted_blk[i] 中找小于 x - add[i] 的最大值
        auto it = lower_bound(sorted_blk[i].begin(), sorted_blk[i].end(), x - add[i]);
        if (it != sorted_blk[i].begin()) {
            --it;
            ans = max(ans, *it + add[i]);
        }
    }

    for (int i = L[bl_r]; i <= r; i++) {
        long long val = a[i] + add[bl_r];
        if (val < x) ans = max(ans, val);
    }

    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
    }

    init();

    while (m--) {
        int op;
        cin >> op;
        if (op == 1) {
            int l, r, v;
            cin >> l >> r >> v;
            range_add(l, r, v);
        } else if (op == 2) {
            int l, r;
            long long x;
            cin >> l >> r >> x;
            cout << query_less(l, r, x) << "\n";
        } else {
            int l, r;
            long long x;
            cin >> l >> r >> x;
            cout << query_pred(l, r, x) << "\n";
        }
    }

    return 0;
}

    7. 整数分块/数论分块¶

    问题¶

    计算 \(\sum_{i=1}^{N} \lfloor N/i \rfloor\)

    核心思想¶

    \(\lfloor N/i \rfloor\) 的取值只有 O(√N) 种。

    • 当 \(i \leq \sqrt{N}\) 时，最多 √N 种取值
    • 当 \(i > \sqrt{N}\) 时，\(\lfloor N/i \rfloor \leq \sqrt{N}\)，最多 √N 种取值

    对于相同的 \(\lfloor N/i \rfloor = k\)，找到最大的 \(j\) 使得 \(\lfloor N/j \rfloor = k\)，则 \(j = \lfloor N/k \rfloor\)。

void bfs(const vector<int>& sources) {
    queue<int> q;
    for (int i = 1; i <= n; i++) dist[i] = 1e9;
    for (int s : sources) {
        dist[s] = 0;
        q.push(s);
    }

        q.pop();
        for (int v : adj[u]) {
            if (dist[v] > dist[u] + 1) {
                dist[v] = dist[u] + 1;
                q.push(v);
            }
        }
    }
}

    return abs(u - v);  // 占位

    auto it1 = lower_bound(pos[x].begin(), pos[x].end(), l);
    auto it2 = upper_bound(pos[x].begin(), pos[x].end(), r);
    return it2 - it1;
}

int query_mode(int l, int r) {
    int bl_l = bl[l], bl_r = bl[r];

    if (bl_l == bl_r || bl_l + 1 == bl_r) {
        // 暴力
        map<int, int> cnt;
        int mode = 0, max_cnt = 0;
        for (int i = l; i <= r; i++) {
            cnt[a[i]]++;
            if (cnt[a[i]] > max_cnt || (cnt[a[i]] == max_cnt && a[i] < mode)) {
                max_cnt = cnt[a[i]];
                mode = a[i];
            }
        }
        return mode;
    }

    int mode = f[bl_l + 1][bl_r - 1];
    int max_cnt = count_in_range(mode, l, r);

    // 检查左散块
    for (int i = l; i <= R[bl_l]; i++) {
        int c = count_in_range(a[i], l, r);
        if (c > max_cnt || (c == max_cnt && a[i] < mode)) {
            max_cnt = c;
            mode = a[i];
        }
    }

    // 检查右散块
    for (int i = L[bl_r]; i <= r; i++) {
        int c = count_in_range(a[i], l, r);
        if (c > max_cnt || (c == max_cnt && a[i] < mode)) {
            max_cnt = c;
            mode = a[i];
        }
    }

    return mode;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
    }

    for (int i = 1; i <= num_blk; i++) {
        L[i] = (i - 1) * B + 1;
        R[i] = min(i * B, n);
    }

    init();

    while (m--) {
        int l, r;
        cin >> l >> r;
        cout << query_mode(l, r) << "\n";
    }

    return 0;
}

    复杂度：预处理 O(N²/B)，查询 O(√N log N)。块大小需要手动调参。

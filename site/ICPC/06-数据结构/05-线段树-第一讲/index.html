    1. 大白话引入¶

    还是那个仓库问题：\(n\) 个货架，每个架子上有零件数。老板随时会：

    1. 改：往某个架子加/减零件
    2. 查：问一段区间 \([l, r]\) 的总零件数、最大值、或者更奇怪的东西（比如"区间里最长连续上升段"）

    树状数组（BIT）能解决"改 + 求和"，但有个致命限制：和可以减回去（区间和 = 前缀和相减），而最大值、gcd、按位或这些信息没有逆运算，树状数组就无能为力了。

    线段树的出现就是为了回答：能不能有一种结构，只要能"把两个相邻区间的答案合并成一个区间"，就能维护？ 答案就是线段树——修改和查询都 \(O(\log n)\)，不要求信息可减，只要求可合并（满足结合律）。

    2. 思维启发：为什么需要线段树¶

    如果我是做题人，我会怎么想？

    先盘点已有工具：

    做法 单点修改 区间查询 短板
    朴素数组 \(O(1)\) \(O(n)\) 查询太慢
    前缀和 \(O(n)\) \(O(1)\) 修改太慢
    ST 表 不能修改 \(O(1)\) 只支持静态
    树状数组 \(O(\log n)\) \(O(\log n)\) 依赖可减性，max/gcd 不行

    前三个都是"偏科生"。树状数组是好学生，但只懂"减法"。如果老板问的是区间最大值呢？ max 没有"逆操作"——知道 [1,5] 的最大值和 [1,3] 的最大值，减不出 [4,5] 的最大值。

    关键转折：我们不追求"从两个前缀推区间"，而是把区间预先切成 O(log n) 块存在树上，查询时拼这些块。只要"两块拼一块"的合并操作是 \(O(1)\) 且满足结合律，什么信息都能维护。

    手算最小例子。\(a = [3, 1, 4, 1, 5, 9, 2, 6]\)（\(n=8\)），把区间对半切：

    3. 结构与基本操作（P3374 风格模板）¶

    3.1 结构：区间二分 + 完全二叉树存储¶

    • 根节点对应 [1, n]，左右儿子各管一半区间，叶子是单点
    • 节点 \(i\) 的左儿子 \(2i\)、右儿子 \(2i+1\)（和堆一样的存储方式）
    • 数组必须开 \(4n\)：递归建树节点总数小于 \(4n\)，这是竞赛圈经验值，开 \(2n\) 会越界（为什么不是 \(2n\)：线段树不一定是完美二叉树，最后一层可能有多余节点）

    3.2 四个固定件¶

    函数 作用 时机
    build 递归建树，叶子赋值，回溯合并 初始化
    update 单点修改，改完一路 pushup 每次修改
    query 区间查询，整段直接返回 每次查询
    pushup 用两个儿子合并出父节点 回溯时

    核心心法：后面所有题只改 pushup（怎么合并）和叶子信息（存什么），框架永远不变。

    #include <bits/stdc++.h>
using namespace std;

const int MAXN = 100005;
int n, m;
long long a[MAXN], t[MAXN << 2];  // 4n 空间：节点 i 的区间和

// pushup：用左右儿子的值合并出父亲
// 为什么回溯时调用：build/update 递归返回时，儿子已经算好，父亲才能合并
void pushup(int p) {
    t[p] = t[p << 1] + t[p << 1 | 1];  // 区间和 = 左儿子 + 右儿子
}

// build：递归建树，O(n)
void build(int p, int l, int r) {
    if (l == r) {           // 叶子：只存一个数
        t[p] = a[l];
        return;
    }
    int m = (l + r) >> 1;   // 二分中点
    build(p << 1, l, m);            // 先建左子树 [l, m]
    build(p << 1 | 1, m + 1, r);    // 再建右子树 [m+1, r]
    pushup(p);              // 子树建完，回溯合并出本节点
}

// update：单点修改 a[pos] += v，O(log n)
void update(int p, int l, int r, int pos, long long v) {
    if (l == r) {           // 找到叶子
        t[p] += v;
        return;
    }
    int m = (l + r) >> 1;
    if (pos <= m) update(p << 1, l, m, pos, v);        // pos 在左半边
    else          update(p << 1 | 1, m + 1, r, pos, v); // pos 在右半边
    pushup(p);              // 回溯时更新路径上的所有祖先
}

// query：查询区间 [ql, qr] 的和，O(log n)
long long query(int p, int l, int r, int ql, int qr) {
    if (ql <= l && r <= qr) return t[p];  // 全覆盖：这个节点就是答案的一部分
    int m = (l + r) >> 1;
    long long res = 0;
    // 只往有交集的儿子递归（别两边都进，会多算）
    if (ql <= m) res += query(p << 1, l, m, ql, qr);
    if (qr >  m) res += query(p << 1 | 1, m + 1, r, ql, qr);
    return res;
}

int main() {
    // HDU 1166 敌兵布阵风格：T 组数据，Add/Sub/Query/End 操作
    int T;
    scanf("%d", &T);
    for (int cas = 1; cas <= T; cas++) {
        scanf("%d", &n);
        for (int i = 1; i <= n; i++) scanf("%lld", &a[i]);
        build(1, 1, n);
        printf("Case %d:\n", cas);
        char op[10];
        while (scanf("%s", op), op[0] != 'E') {  // End 结束本组
            if (op[0] == 'A') {       // Add i j：a[i] += j
                int i; long long j;
                scanf("%d%lld", &i, &j);
                update(1, 1, n, i, j);
            } else if (op[0] == 'S') { // Sub i j：a[i] -= j
                int i; long long j;
                scanf("%d%lld", &i, &j);
                update(1, 1, n, i, -j);
            } else {                   // Query i j：区间 [i,j] 和
                int i, j;
                scanf("%d%d", &i, &j);
                printf("%lld\n", query(1, 1, n, i, j));
            }
        }
    }
    return 0;
}

    复杂度：build \(O(n)\)；update/query 每层只走一条路径或访问 \(O(1)\) 个整段，\(O(\log n)\)。\(n=10^5\) 时深度约 17 层。

    #include <bits/stdc++.h>
using namespace std;

const int MAXN = 100005;
const long long MOD = 1e9 + 7;
int n, m;
long long sum[MAXN << 2], sum2[MAXN << 2];  // sum：Σx；sum2：Σx²

// 快速幂求逆元：a^(p-2) mod p（费马小定理，p 是质数）
long long modpow(long long a, long long b) {
    long long r = 1;
    while (b) {
        if (b & 1) r = r * a % MOD;
        a = a * a % MOD;
        b >>= 1;
    }
    return r;
}

// pushup：两个信息分别合并，互不干扰
void pushup(int p) {
    sum[p]  = (sum[p << 1]  + sum[p << 1 | 1]) % MOD;
    sum2[p] = (sum2[p << 1] + sum2[p << 1 | 1]) % MOD;
}

void build(int p, int l, int r) {
    if (l == r) {
        long long x;
        scanf("%lld", &x);
        sum[p]  = x % MOD;
        sum2[p] = x * x % MOD;  // 叶子直接存平方
        return;
    }
    int m = (l + r) >> 1;
    build(p << 1, l, m);
    build(p << 1 | 1, m + 1, r);
    pushup(p);
}

void update(int p, int l, int r, int pos, long long v) {
    if (l == r) {
        sum[p]  = v % MOD;
        sum2[p] = v * v % MOD;
        return;
    }
    int m = (l + r) >> 1;
    if (pos <= m) update(p << 1, l, m, pos, v);
    else          update(p << 1 | 1, m + 1, r, pos, v);
    pushup(p);
}

// 返回 {区间和, 区间平方和}
pair<long long, long long> query(int p, int l, int r, int ql, int qr) {
    if (ql <= l && r <= qr) return {sum[p], sum2[p]};
    int m = (l + r) >> 1;
    long long s = 0, s2 = 0;
    if (ql <= m) {
        auto L = query(p << 1, l, m, ql, qr);
        s = (s + L.first) % MOD; s2 = (s2 + L.second) % MOD;
    }
    if (qr > m) {
        auto R = query(p << 1 | 1, m + 1, r, ql, qr);
        s = (s + R.first) % MOD; s2 = (s2 + R.second) % MOD;
    }
    return {s, s2};
}

int main() {
    scanf("%d%d", &n, &m);
    build(1, 1, n);
    while (m--) {
        int op, x, y;
        scanf("%d%d%d", &op, &x, &y);
        if (op == 1) update(1, 1, n, x, y);       // C x y：a[x] = y
        else {
            // Q l r：输出方差 D = (n·Σx² - (Σx)²) / n² 对 MOD 取模
            auto [s, s2] = query(1, 1, n, x, y);
            long long len = y - x + 1;
            // 分子 = n·Σx² - (Σx)²，注意取模后可能是负数，+MOD 修正
            long long num = (len % MOD * s2 % MOD - s * s % MOD + MOD) % MOD;
            long long inv = modpow(len % MOD * (len % MOD) % MOD, MOD - 2);
            printf("%lld\n", num * inv % MOD);
        }
    }
    return 0;
}

    坑：方差分子必须先算 n·Σx² - (Σx)² 再乘 n² 的逆元。如果先算 \(E(x^2)\) 和 \(E(x)^2\) 再相减，中间会有分数，模运算会出问题。

    5. 结合律：线段树能维护什么¶

    判定标准：两个相邻区间的答案能否 \(O(1)\) 合并成整个区间的答案 → 能，就能用线段树维护。

    信息 单位元 说明
    和 / 异或 0 / 0 可减，树状数组也能做
    积 1 不可减（含 0 无逆元），但可合并
    max / min / gcd / 按位或 \(-\infty\) / \(+\infty\) / 0 / 0 不可减，但可合并
    最大子段和 四元组 要打包四个值
    矩阵乘法 单位矩阵 合并 = 矩阵乘
    摩尔投票 \((0, 0)\) 候选 + 票数
    反例 — 中位数、众数、单独的平均值（两个区间合并不出整体答案）

    #include <bits/stdc++.h>
using namespace std;

const int MAXN = 500005;
const long long NEG = -1e18;
int n, m;
long long a[MAXN];

struct Node {          // 每个节点打包四个值
    long long sum;     // 整个区间和
    long long lmax;    // 从左端开始的最大连续和
    long long rmax;    // 到右端结束的最大连续和
    long long best;    // 区间内最大子段和（答案）
};

Node t[MAXN << 2];

Node merge(Node L, Node R) {
    Node res;
    res.sum  = L.sum + R.sum;
    // 左起：要么停在 L 内（L.lmax），要么穿完 L 再进 R（L.sum + R.lmax）
    res.lmax = max(L.lmax, L.sum + R.lmax);
    res.rmax = max(R.rmax, R.sum + L.rmax);
    // 答案：在 L、在 R、或跨中点的 L.rmax + R.lmax
    res.best = max({L.best, R.best, L.rmax + R.lmax});
    return res;
}

void pushup(int p) { t[p] = merge(t[p << 1], t[p << 1 | 1]); }

void build(int p, int l, int r) {
    if (l == r) {
        t[p] = {a[l], a[l], a[l], a[l]};  // 叶子：四个值都一样
        return;
    }
    int m = (l + r) >> 1;
    build(p << 1, l, m);
    build(p << 1 | 1, m + 1, r);
    pushup(p);
}

void update(int p, int l, int r, int pos, long long v) {
    if (l == r) { t[p] = {v, v, v, v}; return; }
    int m = (l + r) >> 1;
    if (pos <= m) update(p << 1, l, m, pos, v);
    else          update(p << 1 | 1, m + 1, r, pos, v);
    pushup(p);
}

// 查询返回合并好的 Node；无交集返回"对合并无影响"的单位元
Node unit() { return {0, NEG, NEG, NEG}; }  // sum=0，其他 -INF：merge 时不会污染

Node query(int p, int l, int r, int ql, int qr) {
    if (ql <= l && r <= qr) return t[p];
    int m = (l + r) >> 1;
    Node res = unit();
    bool has = false;
    if (ql <= m) { res = query(p << 1, l, m, ql, qr); has = true; }
    if (qr > m) {
        Node R = query(p << 1 | 1, m + 1, r, ql, qr);
        res = has ? merge(res, R) : R;  // 注意：左边没取到时要直接赋值
    }
    return res;
}

int main() {
    scanf("%d%d", &n, &m);
    for (int i = 1; i <= n; i++) scanf("%lld", &a[i]);
    build(1, 1, n);
    while (m--) {
        int op, x, y;
        scanf("%d%d%d", &op, &x, &y);
        if (op == 1) {
            if (x > y) swap(x, y);  // P4513 的查询可能 l > r，先交换
            printf("%lld\n", query(1, 1, n, x, y).best);
        } else update(1, 1, n, x, y);
    }
    return 0;
}

    坑：查询时若左半无交集，不能把单位元直接 merge 右半——unit().lmax = -INF 会把右半的 lmax 拉低。正确写法是"第一个取到的结果直接赋值，之后用 merge 拼接"（上面的 has 变量）。

    #include <bits/stdc++.h>
using namespace std;

const int MAXN = 1000005;
string s;
int n, m;

struct Node { int open, close, ans; };
Node t[MAXN << 2];

Node merge(Node L, Node R) {
    int x = min(L.open, R.close);   // 左的 '(' 配右的 ')'，能配多少配多少
    return {L.open + R.open - x,    // 剩下的 '('：L 的 + R 的 - 被配走的
            L.close + R.close - x,  // 剩下的 ')'
            L.ans + R.ans + x};     // 已配对数累加
}

void build(int p, int l, int r) {
    if (l == r) {
        if (s[l] == '(') t[p] = {1, 0, 0};  // 叶子：一个 '('
        else             t[p] = {0, 1, 0};  // 叶子：一个 ')'
        return;

    int m = (l + r) >> 1;
    build(p << 1, l, m);
    build(p << 1 | 1, m + 1, r);
    t[p] = merge(t[p << 1], t[p << 1 | 1]);
}

    if (qr > m) {

        res = has ? merge(res, R) : R;
    }
    return res;
}

    6. 动态开点¶

    动机¶

    值域太大（\(10^9\)）时，普通线段树开 \(4 \times 10^9\) 个节点直接爆内存。离散化可以压缩，但不支持在线插入新值（要先把所有可能值收集齐）。

    #include <bits/stdc++.h>
using namespace std;

const int MAXN = 500005;
const int MAXV = 1e9;
int n, tot;                 // tot：已创建节点数
struct Node { int lc, rc; long long sum; } tr[MAXN * 40];
// 空间为什么是 Q*40：每次插入沿值域走 log2(1e9)≈30 层，Q 次最多 Q*30 个新节点

// 单点插入：在值域 [l,r] 的线段树上给 pos 加 v，访问到哪建到哪
void upd(int &p, int l, int r, int pos, int v) {
    if (!p) p = ++tot;      // 第一次访问到这个节点：现场创建
    tr[p].sum += v;
    if (l == r) return;
    int m = (l + r) >> 1;
    if (pos <= m) upd(tr[p].lc, l, m, pos, v);    // 递归时引用传递，自动创建
    else          upd(tr[p].rc, m + 1, r, pos, v);
}

// 查询值域 [ql, qr] 的个数和
long long query(int p, int l, int r, int ql, int qr) {
    if (!p || qr < l || r < ql) return 0;   // 空节点或完全无交集
    if (ql <= l && r <= qr) return tr[p].sum;
    int m = (l + r) >> 1;
    return query(tr[p].lc, l, m, ql, qr) + query(tr[p].rc, m + 1, r, ql, qr);
}

int main() {
    // P1908 逆序对：值域 1e9 不用离散化，动态开点直接做
    scanf("%d", &n);
    int root = 0;
    long long ans = 0;
    for (int i = 0; i < n; i++) {
        int x;
        scanf("%d", &x);
        // 已经插入的、值域在 [x+1, 1e9] 的个数 = 比 x 大的 = 逆序对
        ans += query(root, 1, MAXV, x + 1, MAXV);
        upd(root, 1, MAXV, x, 1);           // 插入 x
    }
    printf("%lld\n", ans);
    return 0;
}

    坑：upd 的参数 int &p 必须传引用，否则递归创建的节点丢了。query 先判 !p 返回 0，避免访问空节点。

    7. 主席树（可持久化线段树）¶

    动机：静态区间第 K 小（P3834）¶

    问 \([l, r]\) 中第 \(k\) 小的数。思维启发：如果对每个前缀 \([1, i]\) 建一棵值域线段树（版本 \(i\) = 前 \(i\) 个数的值域统计），那么 \([l, r]\) 的统计 = 版本 \(r\) − 版本 \(l-1\)，在差值树上二分就能找第 \(k\) 小。

    朴素建 \(n\) 棵树空间 \(O(n^2)\) 爆掉。可持久化把它压到 \(O(n \log n)\)：新版本只沿修改路径新建 \(O(\log V)\) 个节点，其余子树直接指回旧版本（共享）。

    #include <bits/stdc++.h>
using namespace std;

const int MAXN = 200005;
int n, m, tot;
int a[MAXN], root[MAXN];  // root[i]：版本 i 的树根
struct Node { int lc, rc, cnt; } tr[MAXN * 25];  // 每个版本新建 logV 个节点

// 在旧版本 pre 的基础上插入 pos，返回新版本根
int upd(int pre, int l, int r, int pos) {
    int p = ++tot;
    tr[p] = tr[pre];        // 复制旧节点（lc/rc/cnt 全部继承，未改的子树直接共享）
    tr[p].cnt++;            // 本节点计数 +1
    if (l == r) return p;
    int m = (l + r) >> 1;
    if (pos <= m) tr[p].lc = upd(tr[pre].lc, l, m, pos);  // 只重建左链
    else          tr[p].rc = upd(tr[pre].rc, m + 1, r, pos);
    return p;
}

// 在"版本 u 减版本 v"的差值树上找第 k 小（u 是 l-1，v 是 r）
int kth(int u, int v, int l, int r, int k) {
    if (l == r) return l;   // 找到答案（离散化后的下标）
    int m = (l + r) >> 1;
    int cntL = tr[tr[v].lc].cnt - tr[tr[u].lc].cnt;  // 左子树里 [l,r] 区间的数
    if (k <= cntL) return kth(tr[u].lc, tr[v].lc, l, m, k);     // 第 k 小在左
    return kth(tr[u].rc, tr[v].rc, m + 1, r, k - cntL);         // 在右，减掉左边
}

int main() {
    scanf("%d%d", &n, &m);
    vector<int> b(n + 1);
    for (int i = 1; i <= n; i++) { scanf("%d", &a[i]); b[i] = a[i]; }
    // 离散化：值域压到 1..n
    sort(b.begin() + 1, b.end());
    b.erase(unique(b.begin() + 1, b.end()), b.end());
    int V = b.size() - 1;
    // 逐版本建树：版本 i = 版本 i-1 + 插入 a[i]
    for (int i = 1; i <= n; i++) {
        int pos = lower_bound(b.begin() + 1, b.end(), a[i]) - b.begin();
        root[i] = upd(root[i - 1], 1, V, pos);
    }
    while (m--) {
        int l, r, k;
        scanf("%d%d%d", &l, &r, &k);
        int idx = kth(root[l - 1], root[r], 1, V, k);
        printf("%d\n", b[idx]);   // 离散化下标映射回原值
    }
    return 0;
}

    易错点：① 版本号写错全盘皆错，查询参数是 (root[l-1], root[r])，顺序别混；② tr[0] 必须是全 0 的空节点（全局数组默认就是），否则版本相减会出错；③ 输出前把离散化下标映射回原值。

    8. 线段树优化建图¶

    CF786B：Legacy¶

    三种连边：① \(u \to v\) ② \(u \to [l, r]\) ③ \([l, r] \to u\)。暴力对区间内每个点连边 \(O(n)\)，最坏 \(O(nm)\) 不可接受。

    核心思想：任何区间 \([l, r]\) 都能拆成 \(O(\log n)\) 个线段树"整段"。把这些整段对应的节点当虚拟节点，区间连边只连虚拟节点：

    • 出树（处理 \(u \to [l,r]\)）：父 → 子连 0 边（从区间出发能下到每个点），叶子 = 原节点
    • 入树（处理 \([l,r] \to u\)）：子 → 父连 0 边（从每个点能上到区间），叶子 = 原节点

    点数 \(O(n)\)、边数 \(O((n+m)\log n)\)，建完图直接跑 Dijkstra。

    9. 本节重点¶

    本节重点： - 线段树 = 区间二分 + 完全二叉树存储，数组开 \(4n\) - 四个固定件：build / update / query / pushup，框架不变，只改节点信息和 pushup - 结合律：可合并就能维护，不要求可减（树状数组做不到的 max/gcd 都能做） - 打包思想：信息不够就补——最大子段和四元组、括号三元组、方差二元组 - 动态开点：值域大时按需建点，空间 \(O(Q \log V)\)，支持在线插入 - 主席树：前缀版本 + 共享子树，静态区间第 K 小，空间 \(O(n \log n)\) - 优化建图：出树 + 入树，区间连边拆成 \(O(\log n)\) 条 - 常见坑： - 单位元别写错（max 用 \(-INF\)，不是 0） - 多区间拼接查询时用 has 标记，别拿单位元硬 merge - 主席树版本号 \((l-1, r)\) 顺序、离散化映射回原值 - 动态开点 upd 参数要传引用，query 先判空节点

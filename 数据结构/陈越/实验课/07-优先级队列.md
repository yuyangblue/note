---

医院急诊科需要根据病人的病情严重程度决定接诊顺序。危重病人（严重程度 1）必须优先于轻症病人（严重程度 5）接诊。优先队列可以完美解决这个问题：优先级最高的元素总是最先出队。

本实验用 C 语言实现基于堆的优先队列，通过控制台菜单交互，支持添加病人、接诊出队、查看等待列表、查看下一个、等待时间统计和科室统计等功能。

|功能模块|子功能|详细说明|
|---|---|---|
|添加病人|入队|输入姓名/严重程度/科室，按优先级插入堆|
|接诊|出队|取出优先级最高的病人（severity 最小）|
|查看等待列表|显示所有病人|按堆顺序列出所有等待病人|
|查看下一个|peek|查看优先级最高的病人但不移除|
|等待时间统计|平均等待|计算所有等待病人的平均到达时间|
|科室统计|按科室分组|统计各科室的等待病人数|

优先队列是一种特殊的队列，出队顺序由元素的优先级决定，而非入队顺序。本实验中，严重程度数字越小，优先级越高。

堆是一棵完全二叉树，用数组存储。小根堆中，父节点的值 <= 子节点的值。堆的数组存储：根节点在下标 1，节点 i 的左孩子为 2i，右孩子为 2i+1。

| 操作  | 方法                         | 时间复杂度    |
| :-- | :------------------------- | :------- |
| 插入  | 放入末尾，向上调整 heapifyUp        | O(log n) |
| 删除  | 根出队，末尾移入根，向下调整 heapifyDown | O(log n) |
| 查看  | 直接返回根                      | O(1)     |

typedef struct {
    int id;              // 病人ID

    int arrivalTime;     // 到达时间

    Patient *arr;   // 堆数组（下标从1开始）
    int size;       // 当前元素数
    int capacity;   // 容量
} Heap;

```c
Patient dequeue(Heap *h) {
    Patient top = h->arr[1];        // 保存根节点（优先级最高）
    h->arr[1] = h->arr[h->size];    // 末尾移入根
    h->size--;
    heapifyDown(h, 1);              // 向下调整
    return top;
}

void heapifyDown(Heap *h, int idx) {
    int n = h->size;
    while (1) {
        int smallest = idx;
        int left = 2 * idx;
        int right = 2 * idx + 1;
        // 找三个节点中优先级最高的（severity最小）
        if (left <= n && h->arr[left].severity < h->arr[smallest].severity)
            smallest = left;
        if (right <= n && h->arr[right].severity < h->arr[smallest].severity)
            smallest = right;
        if (smallest != idx) {
            swap(&h->arr[idx], &h->arr[smallest]);
            idx = smallest;
        } else {
            break;
        }
    }
}
```

```c
void heapifyUp(Heap *h, int idx) {
    // idx 从插入位置开始，向上追溯
    while (idx > 1) {                      // 直到根节点
        int parent = idx / 2;              // 父节点下标
        if (h->arr[idx].severity < h->arr[parent].severity) {
            // 当前节点优先级更高（severity更小），与父交换
            swap(&h->arr[idx], &h->arr[parent]);
            idx = parent;                  // 继续检查父节点
        } else {
            break;                          // 堆性质满足，停止
        }
    }
}
```

```c
void heapifyDown(Heap *h, int idx) {
    int n = h->size;
    while (1) {
        int smallest = idx;                 // 假设当前节点优先级最高
        int left = 2 * idx;                 // 左孩子
        int right = 2 * idx + 1;            // 右孩子

        if (left <= n && h->arr[left].severity < h->arr[smallest].severity)
            smallest = left;                // 左孩子更优先
        if (right <= n && h->arr[right].severity < h->arr[smallest].severity)
            smallest = right;               // 右孩子更优先

        if (smallest != idx) {
            swap(&h->arr[idx], &h->arr[smallest]);  // 交换
            idx = smallest;                          // 继续向下
        } else {
            break;                                   // 堆性质满足
        }
    }
}
```

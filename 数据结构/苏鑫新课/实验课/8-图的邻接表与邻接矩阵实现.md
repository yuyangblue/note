- **二、实验介绍**

---

图由顶点集合和边集合组成，用于表示对象之间的任意关系。邻接矩阵使用 VxV 的二维数组存储边信息，适合稠密图；邻接表为每个顶点维护一个链表存储其邻接点，适合稀疏图。DFS 和 BFS 是两种最基本的图遍历算法，分别使用栈（递归）和队列实现。

本实验实现一个基于 C 语言的图的存储与遍历系统，同时支持邻接矩阵和邻接表两种结构，提供交互式菜单操作，支持顶点和边的增删、DFS/BFS 遍历和两种结构的对比展示。

```c
typedef struct {
    int matrix[MAX][MAX];  /* 邻接矩阵，0/1表示是否有边 */
    char *names[MAX];      /* 顶点名称数组 */
    int n;                 /* 当前顶点数 */
} AdjMatrix;
```

|字段|说明|
|:--|:--|
|matrix[MAX][MAX]|上三角和下三角对称，matrix[i][j]=1 表示顶点i和j相连|
|names[MAX]|存储每个顶点的名称字符串指针|
|n|当前图中的顶点数量|

```c
typedef struct EdgeNode {
    int dest;              /* 目标顶点编号 */
    struct EdgeNode *next; /* 下一个邻接点 */
} EdgeNode;

typedef struct {
    EdgeNode *head;        /* 链表头指针 */
    char *name;            /* 顶点名称 */
} AdjListVertex;

typedef struct {
    AdjListVertex vertices[MAX];  /* 顶点数组 */
    int n;                        /* 当前顶点数 */
} AdjList;
```

|结构体|字段|说明|
|:--|:--|:--|
|EdgeNode|dest, next|邻接点链表节点，dest是目标顶点编号|
|AdjListVertex|head, name|每个顶点对应一个链表头指针|
|AdjList|vertices[], n|顶点数组和当前数量|

```c
AdjMatrix* createMatrix() {
    AdjMatrix *g = malloc(sizeof(AdjMatrix));
    g->n = 0;
    for (int i = 0; i < MAX; i++)
        for (int j = 0; j < MAX; j++)
            g->matrix[i][j] = 0;  // 全部初始化为"无边"
    return g;
}

void addEMatrix(AdjMatrix *g, int u, int v) {
    g->matrix[u][v] = 1;  // u到v有边
    g->matrix[v][u] = 1;  // v到u有边（无向图双向添加）
}

void delVMatrix(AdjMatrix *g, int idx) {
    free(g->names[idx]);  // 释放名称内存
    // 从idx开始，将后面的行和列前移
    for (int i = idx; i < g->n-1; i++) {
        g->names[i] = g->names[i+1];       // names前移
        for (int j = 0; j < g->n; j++)
            g->matrix[i][j] = g->matrix[i+1][j];  // 行前移
    }
    for (int i = 0; i < g->n; i++)
        for (int j = idx; j < g->n-1; j++)
            g->matrix[i][j] = g->matrix[i][j+1];  // 列前移
    g->n--;
}
```

```c
void addEList(AdjList *g, int u, int v) {
    // ===== 第一部分：在u的链表中插入v =====
    // 因为是无向图，所以需要双向添加

    EdgeNode *e = malloc(sizeof(EdgeNode));
    // 第1行：创建新的边节点
    // malloc 分配 EdgeNode 结构体大小的内存
    // 返回的内存地址赋值给 e

    e->dest = v;
    // 第2行：设置目标顶点编号为 v
    // 表示这条边连接到顶点 v

    e->next = g->vertices[u].head;
    // 第3行：新节点的next指向u的当前链表头
    // 这是头插法的关键步骤
    // 新节点 -> 原第一个邻接点

    g->vertices[u].head = e;
    // 第4行：u的链表头指向新节点
    // 新节点成为u的第一个邻接点
    // 完成头插：新节点插入链表头部

    // ===== 第二部分：在v的链表中插入u（双向）=====
    e = malloc(sizeof(EdgeNode));
    e->dest = u;
    e->next = g->vertices[v].head;
    g->vertices[v].head = e;
    // 同理在v的链表头部插入u
    // 无向图需要双向添加，确保对称性
}
```

```c
void bfsMatrix(AdjMatrix *g, int start) {
    // ---- 第1部分：初始化 ----
    int q[MAX], f = 0, r = 0;
    // q[] 用作队列，存储待访问的顶点编号
    // f = front 队首指针，指向下一个要出队的元素
    // r = rear  队尾指针，指向下一个入队的位置
    // f == r 表示队列为空
    // 数组模拟队列，使用 FIFO 先进先出特性

    for (int i = 0; i < g->n; i++) visited[i] = 0;
    // 清空 visited 标记数组
    // visited[i] = 0 表示顶点i未被访问
    // visited[i] = 1 表示顶点i已被访问（或已入队）

    // ---- 第2部分：起点入队 ----
    visited[start] = 1;
    // 标记起点为"已访问"
    // BFS 必须在入队时标记，而非出队时！
    // 如果在出队时才标记，同一顶点可能被多次入队

    q[r++] = start;
    // 起点入队：q[0] = start, r 变为 1
    // 入队操作：先赋值 q[r]，再 r++

    // ---- 第3部分：BFS主循环 ----
    while (f < r) {
        // 队列不为空时继续循环
        // f < r 表示队列中还有元素

        int v = q[f++];
        // 出队：取出队首顶点编号
        // 出队操作：先取 q[f]，再 f++
        // 出队的顶点就是要访问的顶点

        printf("%s ", g->names[v]);
        // 访问当前出队的顶点
        // 输出顶点名称

        // ---- 第4部分：邻接点入队 ----
        for (int i = 0; i < g->n; i++) {
            // 扫描所有顶点，找出当前顶点的邻接点

            if (g->matrix[v][i] && !visited[i]) {
                // 条件1: g->matrix[v][i] == 1
                //       表示顶点v和i之间有边
                // 条件2: !visited[i]
                //       表示顶点i尚未被访问

                visited[i] = 1;
                // 入队时立即标记！
                // 这是BFS的关键点，防止同一顶点被多次入队
                // 如果出队时才标记，会导致：
                //   顶点A的邻接点B和C都会将D入队
                //   D会被入队两次！

                q[r++] = i;
                // 将未访问的邻接点入队
                // 等待后续出队访问
            }
        }
    }
}
// 输出: A B C D E
```

    g->vertices[u].head = e;                  // 头插法

void delVMatrix(AdjMatrix *g, int idx) {
    free(g->names[idx]);               // 释放名称
    for (int i = idx; i < g->n-1; i++) {
        g->names[i] = g->names[i+1];   // 名称前移
        for (int j = 0; j < g->n; j++)
            g->matrix[i][j] = g->matrix[i+1][j];  // 行前移
    }
    for (int i = 0; i < g->n; i++)
        for (int j = idx; j < g->n-1; j++)
            g->matrix[i][j] = g->matrix[i][j+1];  // 列前移
    g->n--;
}

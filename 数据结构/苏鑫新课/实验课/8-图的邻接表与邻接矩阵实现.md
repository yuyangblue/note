- **二、实验介绍**

---

图由顶点集合和边集合组成，用于表示对象之间的任意关系。邻接矩阵使用 VxV 的二维数组存储边信息，适合稠密图；邻接表为每个顶点维护一个链表存储其邻接点，适合稀疏图。DFS 和 BFS 是两种最基本的图遍历算法，分别使用栈（递归）和队列实现。

本实验实现一个基于 C 语言的图的存储与遍历系统，同时支持邻接矩阵和邻接表两种结构，提供交互式菜单操作，支持顶点和边的增删、DFS/BFS 遍历和两种结构的对比展示。

- **四、数据模型定义（全局通用）**

---

```c
typedef struct {
    int matrix[MAX][MAX];  /* 邻接矩阵，0/1表示是否有边 */
    char *names[MAX];      /* 顶点名称数组 */
    int n;                 /* 当前顶点数 */
} AdjMatrix;
```

|字段|说明|
|:--|:--|
|matrix[MAX][MAX]|上三角和下三角对称，matrix[i][j]=1 表示顶点i和j相连|
|names[MAX]|存储每个顶点的名称字符串指针|
|n|当前图中的顶点数量|

```c
typedef struct EdgeNode {
    int dest;              /* 目标顶点编号 */
    struct EdgeNode *next; /* 下一个邻接点 */
} EdgeNode;

typedef struct {
    EdgeNode *head;        /* 链表头指针 */
    char *name;            /* 顶点名称 */
} AdjListVertex;

typedef struct {
    AdjListVertex vertices[MAX];  /* 顶点数组 */
    int n;                        /* 当前顶点数 */
} AdjList;
```

|结构体|字段|说明|
|:--|:--|:--|
|EdgeNode|dest, next|邻接点链表节点，dest是目标顶点编号|
|AdjListVertex|head, name|每个顶点对应一个链表头指针|
|AdjList|vertices[], n|顶点数组和当前数量|

```c
AdjMatrix* createMatrix() {
    AdjMatrix *g = malloc(sizeof(AdjMatrix));
    g->n = 0;
    for (int i = 0; i < MAX; i++)
        for (int j = 0; j < MAX; j++)
            g->matrix[i][j] = 0;  // 全部初始化为"无边"
    return g;
}

void addEMatrix(AdjMatrix *g, int u, int v) {
    g->matrix[u][v] = 1;  // u到v有边
    g->matrix[v][u] = 1;  // v到u有边（无向图双向添加）
}

void delVMatrix(AdjMatrix *g, int idx) {
    free(g->names[idx]);  // 释放名称内存
    // 从idx开始，将后面的行和列前移
    for (int i = idx; i < g->n-1; i++) {
        g->names[i] = g->names[i+1];       // names前移
        for (int j = 0; j < g->n; j++)
            g->matrix[i][j] = g->matrix[i+1][j];  // 行前移
    }
    for (int i = 0; i < g->n; i++)
        for (int j = idx; j < g->n-1; j++)
            g->matrix[i][j] = g->matrix[i][j+1];  // 列前移
    g->n--;
}
```

```c
void addEList(AdjList *g, int u, int v) {
    // ===== 第一部分：在u的链表中插入v =====
    // 因为是无向图，所以需要双向添加

    EdgeNode *e = malloc(sizeof(EdgeNode));
    // 第1行：创建新的边节点
    // malloc 分配 EdgeNode 结构体大小的内存
    // 返回的内存地址赋值给 e

    e->dest = v;
    // 第2行：设置目标顶点编号为 v
    // 表示这条边连接到顶点 v

    e->next = g->vertices[u].head;
    // 第3行：新节点的next指向u的当前链表头
    // 这是头插法的关键步骤
    // 新节点 -> 原第一个邻接点

    g->vertices[u].head = e;
    // 第4行：u的链表头指向新节点
    // 新节点成为u的第一个邻接点
    // 完成头插：新节点插入链表头部

    // ===== 第二部分：在v的链表中插入u（双向）=====
    e = malloc(sizeof(EdgeNode));
    e->dest = u;
    e->next = g->vertices[v].head;
    g->vertices[v].head = e;
    // 同理在v的链表头部插入u
    // 无向图需要双向添加，确保对称性
}
```

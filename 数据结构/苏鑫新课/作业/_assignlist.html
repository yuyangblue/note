const int MAX = 100000;

bool visits[MAX];
int matrix[MAX][MAX];
 
 
深度优先算法是
void dfs(int root){
    visits[root] = 1;
    cout << root << " ";
    for(int i=0; i<MAX; i++){
        if(matrix[root][i] && !visits[i]){
            dfs(i);
        }
    }
}
时间复杂度是log(n*n)，每次遍历一整行
 
广度优先算法是
void bfs(int root){
    queue<int> q;
    q.push(root);
    visits[root] = 1;
    while(!q.empty()){
        int now = q.front();
        q.pop();
        cout << now << " ";
        for(int i=0; i<MAX; i++){
            if(matrix[now][i] && !visits[i]){
                visits[i] = 1;
                q.push(i);
            }
        }
    }
}
时间复杂度也是log(n*n)，同样每次遍历一整行

- **二、实验介绍**

---

表达式求值是编译器和计算器中的基础功能。人类习惯的中缀表达式（如 `(3+4)*5`）需要括号来标识运算优先级，而计算机更方便处理后缀表达式（如 `3 4 + 5 *`）。后缀表达式可以通过栈直接求值，也可以构建为表达式树，进而支持多种形式的输出和操作。

本实验用 C 语言实现后缀表达式到表达式二叉树的构建，通过控制台菜单交互，支持表达式输入、三种表达式显示、代入求值、变量替换和树结构展示等功能。

|功能模块|子功能|详细说明|
|---|---|---|
|输入后缀表达式|构建表达式树|用户输入空格分隔的后缀表达式，构建二叉树|
|显示三种表达式|中缀/前缀/后缀|通过三种遍历方式输出表达式|
|代入求值|计算表达式的值|支持无变量和含变量的表达式求值|
|变量替换|替换变量为数值|将表达式中的指定变量替换为数值常量|
|显示树结构|缩进层次输出|以缩进形式展示二叉树的结构|

表达式树是一棵二叉树，其中：

- **叶子节点**：操作数（数字常量或变量）
    
- **内部节点**：运算符（+、-、*、/）
    
- 左子树和右子树分别代表运算符的两个操作数
    

利用栈实现：遍历后缀表达式的每个 token，如果是操作数则创建节点入栈，如果是运算符则弹出两个节点作为右左孩子，创建运算符节点入栈。最后栈顶即为表达式树的根节点。

|表达式形式|遍历方式|示例 `(3+4)*5`|
|:--|:--|:--|
|后缀|后序遍历|`3 4 + 5 *`|
|中缀|中序遍历|`(3 + 4) * 5`|
|前缀|前序遍历|`* + 3 4 5`|

```c
typedef struct ExprNode {
    char type;           // 'n':数字 'o':运算符 'v':变量
    double val;          // 数字节点的值
    char op;             // 运算符节点的运算符 (+ - * /)
    char varName[16];    // 变量节点的变量名
    struct ExprNode *left, *right;  // 左右孩子
} ExprNode;

typedef ExprNode* ExprTree;
```

|成员|说明|
|:--|:--|
|`type`|节点类型：'n'数字、'o'运算符、'v'变量|
|`val`|type='n' 时有效，存储数字值|
|`op`|type='o' 时有效，存储运算符|
|`varName`|type='v' 时有效，存储变量名|
|`left/right`|左右孩子指针|

```c
static ExprNode* createNode(char type) {
    ExprNode *n = (ExprNode*)malloc(sizeof(ExprNode));
    n->type = type;
    n->op = 0;
    n->val = 0.0;
    n->varName[0] = '\0';
    n->left = n->right = NULL;
    return n;
}
```

```c
ExprTree buildFromPostfix(const char *postfix) {
    ExprNode *stack[256];
    int top = 0;
    const char *p = postfix;
    char buf[64];

    while (*p) {
        if (isspace(*p)) { p++; continue; }

        if (isalpha(*p)) {             // 变量：字母开头
            // 读取变量名
            ExprNode *n = createNode('v');
            strcpy(n->varName, buf);
            stack[top++] = n;
        } else if (isdigit(*p) || *p == '.') {  // 数字
            // 读取数字
            ExprNode *n = createNode('n');
            n->val = atof(buf);
            stack[top++] = n;
        } else if (*p == '+' || *p == '-' || *p == '*' || *p == '/') {
            ExprNode *n = createNode('o');
            n->op = *p;
            n->right = stack[--top];    // 注意：先右后左
            n->left = stack[--top];
            stack[top++] = n;
            p++;
        } else { p++; }
    }
    return stack[0];  // 栈顶为根节点
}
```

**构建过程**：

|Token|操作|栈内容|
|:--|:--|:--|
|`3`|创建数字节点入栈|`[3]`|
|`4`|创建数字节点入栈|`[3, 4]`|
|`+`|弹出 4(右) 3(左)，创建+节点入栈|`[+]`|
|`5`|创建数字节点入栈|`[+, 5]`|
|`*`|弹出 5(右) +(左)，创建*节点入栈|`[*]`|

```c
// 中序遍历（中缀表达式）
void printInfix(ExprTree t) {
    if (!t) return;
    if (t->type == 'n') printf("%d", (int)t->val);
    else if (t->type == 'v') printf("%s", t->varName);
    else {
        int needParen = (t->op == '+' || t->op == '-');
        // 左子树
        if (needParen && t->left && t->left->type == 'o'
            && (t->left->op=='+'||t->left->op=='-'))
            { printf("("); printInfix(t->left); printf(")"); }
        else printInfix(t->left);
        printf(" %c ", t->op);
        // 右子树
        if (needParen && t->right && t->right->type == 'o'
            && (t->right->op=='+'||t->right->op=='-'))
            { printf("("); printInfix(t->right); printf(")"); }
        else printInfix(t->right);
    }
}

// 前序遍历（前缀表达式）
void printPrefix(ExprTree t) {
    if (!t) return;
    if (t->type == 'n') printf("%d", (int)t->val);
    else if (t->type == 'v') printf("%s", t->varName);
    else { printf("%c ", t->op); printPrefix(t->left); printf(" "); printPrefix(t->right); }
}

// 后序遍历（后缀表达式）
void printPostfix(ExprTree t) {
    if (!t) return;
    printPostfix(t->left); printf(" ");
    printPostfix(t->right); printf(" ");
    if (t->type == 'n') printf("%d", (int)t->val);
    else if (t->type == 'v') printf("%s", t->varName);
    else printf("%c", t->op);
}
```

```c
double evaluate(ExprTree t, const char *vars[], double vals[], int vn) {
    if (!t) return 0.0;
    if (t->type == 'n') return t->val;
    if (t->type == 'v') {
        for (int i = 0; i < vn; i++)
            if (strcmp(t->varName, vars[i]) == 0)
                return vals[i];
        return 0.0;
    }
    // 后序遍历求值
    double l = evaluate(t->left, vars, vals, vn);
    double r = evaluate(t->right, vars, vals, vn);
    switch (t->op) {
        case '+': return l + r;
        case '-': return l - r;
        case '*': return l * r;
        case '/': return (r != 0.0) ? l / r : 0.0;
        default: return 0.0;
    }
}
```

缩进层级表示树的深度，运算符节点有子节点，数字节点没有子节点。

```c
void printTree(ExprTree t, int depth) {
    if (!t) return;
    for (int i = 0; i < depth; i++) printf("  ");  // 缩进
    if (t->type == 'n') printf("%d\n", (int)t->val);
    else if (t->type == 'v') printf("%s\n", t->varName);
    else {
        printf("%c\n", t->op);
        printTree(t->left, depth + 1);   // 递归左子树
        printTree(t->right, depth + 1);  // 递归右子树
    }
}
```

```c
ExprTree substitute(ExprTree t, const char *var, double val) {
    if (!t) return NULL;
    if (t->type == 'v' && strcmp(t->varName, var) == 0) {
        ExprNode *n = createNode('n');  // 创建数字节点替换
        n->val = val;
        return n;
    }
    // 递归复制树
    ExprNode *n = createNode(t->type);
    if (t->type == 'n') n->val = t->val;
    else if (t->type == 'v') strcpy(n->varName, t->varName);
    else {
        n->op = t->op;
        n->left = substitute(t->left, var, val);
        n->right = substitute(t->right, var, val);
    }
    return n;
}
```

```c
void destroyTree(ExprTree t) {
```

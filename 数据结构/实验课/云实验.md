字符串匹配问题：给定一个主串 T 和一个模式串 P，在主串中查找模式串第一次出现的位置。BF 算法是最直观的方法，依次比较每个可能的位置，最坏时间复杂度 O(n*m)。KMP 算法通过预处理模式串构建 next 数组，利用匹配失败时的局部匹配信息跳过不必要的比较，时间复杂度优化到 O(n+m)。

本实验实现一个基于 C 语言的字符串匹配算法对比系统，支持交互式菜单操作，提供 BF 朴素匹配、KMP 匹配、next 数组构建、匹配过程逐步演示和性能对比等功能。

BF 算法（Brute Force，暴力匹配）是最直接的字符串匹配方法。它从主串的每个位置开始，尝试匹配整个模式串。

**算法思想**：用 i 指向主串的当前位置，用 j 指向模式串的当前位置。从主串的第 0 个字符开始，将模式串与主串对齐，逐个字符比较。如果全部匹配则返回位置；如果某字符不匹配，则将模式串右移一位（i回溯），重新开始比较。

```c
int bfMatch(char *t, char *p) {
    // 参数 t: 主串（文本串）
    // 参数 p: 模式串（要查找的字符串）
    // 返回值: p 在 t 中第一次出现的位置
    //         如果未找到则返回 -1

    int n = strlen(t);  // strlen(t) 计算主串 t 的长度
    // 例如 t="ABC", strlen(t)=3
    // 但不包括结尾的 '\0' 字符

    int m = strlen(p);  // strlen(p) 计算模式串 p 的长度
    // 例如 p="AB", strlen(p)=2

    // 外层循环：遍历主串的每个可能的起始位置
    // i 表示当前模式串与主串对齐的位置
    // i 从 0 到 n-m（最后一个可能的起始位置）
    // 例如 n=10, m=3, 则 i 从 0 到 7
    for (int i = 0; i <= n - m; i++) {

        int j;
        // 内层循环：从当前位置 i 开始，逐个比较模式串的字符
        // j 表示模式串中的位置
        for (j = 0; j < m; j++) {

            // t[i+j] 是主串中第 i+j 个字符
            // p[j] 是模式串中第 j 个字符
            if (t[i + j] != p[j]) {
                // 如果对应位置的字符不相等
                break;   // 跳出内层循环，匹配失败
                // break 只会跳出 for(j) 循环
                // 不会跳出 for(i) 循环
            }
        }

        // 执行到这里有两种情况：
        // 情况1: j == m，说明内层循环完整执行了
        //        m 个字符都匹配成功
        // 情况2: j < m，说明中途 break 了
        //        有字符不匹配

        if (j == m) {
            // 所有字符都匹配成功
            // j==m 意味着模式串完全匹配
            return i;   // 返回匹配的起始位置
            // 程序会在找到第一个匹配后立即返回
            // 不会继续往后查找
        }

        // 如果 j != m，说明本次匹配失败
        // 外层循环继续，i 自增
        // 尝试从主串的下一个位置开始匹配
    }

    // 执行到这里说明所有可能的位置都试过了
    // 都没有找到完整的匹配
    return -1;   // 返回 -1 表示未找到
}
```

**BF回溯过程**（主串="ABABD", 模式串="ABD"）：

|步骤|i|j|比较|t[i+j]==p[j]?|操作|
|:-:|:-:|:-:|:--|:--|:--|
|1|0|0|A==A|相等|j=1|
|1|0|1|B==B|相等|j=2|

|3|2|0|A==A|相等|j=1|
|3|2|1|B==B|相等|j=2|
|3|2|2|D==D|相等|j=3 → 匹配!|

```c
void bfDebug(char *t, char *p) {
    int n = strlen(t), m = strlen(p);
    // 获取主串和模式串的长度

    printf("  BF匹配过程:\\n");

    for (int i = 0; i <= n - m; i++) {
        // 外层循环：遍历每个起始位置

        int j;
        for (j = 0; j < m; j++)
            // 内层循环：比较模式串的每个字符
            if (t[i + j] != p[j]) break;

        // 输出当前步骤的比较结果
        printf("    位置%d: \"", i);
        // 输出位置编号

        for (int k = 0; k < m; k++)
            printf("%c", t[i + k]);
        // 输出主串中从 i 开始的 m 个字符
        // 这 m 个字符就是当前与模式串对齐的部分

        printf("\" %s\\n", j == m ? "MATCH!" : "fail");
        // 如果 j==m 表示完全匹配，输出 MATCH!
        // 否则输出 fail

        if (j == m) {
            // 找到匹配
            printf("  结果: 位置 %d\\n", i);
            return;
        }
    }
    printf("  结果: 未找到\\n");
}
```

```c
void buildNext(char *p, int *next) {
    // 参数 p: 模式串
    // 参数 next: 输出的 next 数组
    // next[j] 表示当 p[j] 匹配失败时，j 应该跳转到的位置
    // next[j] = k 意味着 p[0..k-1] = p[j-k..j-1]
    // 即前 j 个字符中，最长相同前后缀的长度为 k

    int m = strlen(p);
    // 模式串的长度

    next[0] = -1;
    // 第一个字符的 next 值固定为 -1
    // 当第一个字符就不匹配时，主串前进，模式串从头开始
    // -1 是一个特殊标记，表示"模式串已退到起点"

    int i = 0;   // i 指向当前正在计算 next 的位置
    int j = -1;  // j 指向前缀的最后一个字符的位置

    // 递推计算 next[1] 到 next[m-1]
    while (i < m - 1) {
        // 循环直到计算出所有位置的 next 值

        if (j == -1 || p[i] == p[j]) {
            // 情况1: j == -1，已退到起点
            //   表示 p[0..i] 中没有相同前后缀
            //   所以 next[i+1] = 0
            // 情况2: p[i] == p[j]
            //   当前字符与前缀字符相等
            //   所以前缀可以扩展一位
            //   next[i+1] = j + 1

            i++;  // 移动到下一个位置
            j++;  // 前缀长度+1
            next[i] = j;
            // next[i+1] = j+1
            // 即新位置的 next 值为当前前缀长度
        } else {
            // p[i] != p[j]
            // 当前字符与前缀字符不相等
            // 需要回溯 j 到更短的前缀
            // 这类似于 KMP 匹配中 j 的跳转

            j = next[j];
            // j 跳转到 next[j] 位置
            // 相当于在前后缀匹配中应用 KMP 思想
            // 递归地找更短的前缀
        }
    }
}
```

```c
int kmpMatch(char *t, char *p) {
    // KMP 匹配算法
    // 核心思想：主串指针 i 永不回溯
    // 模式串指针 j 根据 next 数组跳转

    int n = strlen(t);  // 主串长度
    int m = strlen(p);  // 模式串长度

    int next[MAX];
    // next 数组，长度至少为 m
    // next[j] 存储当 p[j] 不匹配时的跳转位置

    buildNext(p, next);
    // 预处理模式串，构建 next 数组
    // 这个预处理只需要 O(m) 时间
    // 在整个匹配过程中只需要做一次

    int i = 0;  // 主串指针（关键：永远不回溯！）
    int j = 0;  // 模式串指针（根据 next 跳转）

    while (i < n && j < m) {
        // 主循环：直到主串读完或模式串匹配完

        if (j == -1 || t[i] == p[j]) {
            // j==-1: 模式串指针已退到起点
            //   相当于从头开始匹配
            // t[i]==p[j]: 当前字符匹配成功

            i++;  // 主串指针前进
            j++;  // 模式串指针前进
        } else {
            // t[i] != p[j]: 当前字符不匹配
            // KMP 核心操作：
            // 主串指针 i 不动（不回溯）
            // 模式串指针 j 根据 next 数组跳转

            j = next[j];
            // j 跳转到 next[j] 位置
            // next[j] 是前 j 个字符的最长相同前后缀长度
            // 这样 p[0..next[j]-1] 已经和 t[i-next[j]..i-1] 匹配
            // 不需要重新比较这些字符
        }
    }

    // 循环结束后有两种情况：
    if (j == m) {
        // j == m 表示模式串的所有字符都匹配成功了
        // 匹配位置为 i - j（当前主串位置减去模式串长度）
        return i - j;
    } else {
        // j < m 表示主串已经读完但模式串还没匹配完
        // 说明主串中不存在与模式串匹配的子串
        return -1;
    }
}
```

```c
int bfMatch(char *t, char *p) {
    int n = strlen(t);           // 主串长度
    int m = strlen(p);           // 模式串长度
    for (int i = 0; i <= n-m; i++) {  // 主串每个位置
        int j;
        for (j = 0; j < m; j++)       // 模式串每个字符
            if (t[i+j] != p[j]) break; // 不匹配则跳出
        if (j == m) return i;          // 完全匹配
    }
    return -1;                         // 未找到
}
```

|strlen(t)|主串长度|获取主串字符数|
|strlen(p)|模式串长度|获取模式串字符数|
|i<=n-m|外层循环|主串最后n-m个位置开始长度不足|

```c
int kmpMatch(char *t, char *p) {
    int n = strlen(t), m = strlen(p);
    int next[MAX];               // next数组
    buildNext(p, next);          // 预处理：构建next数组

    int i = 0, j = 0;           // i主串指针, j模式串指针
    while (i < n && j < m) {
        if (j == -1 || t[i] == p[j]) { i++; j++; }
        else j = next[j];        // KMP核心：j跳转，i不动
    }
    return (j == m) ? i - j : -1;
}
```

|:--|:--|:--|---|---|
|buildNext(p,next)|构建next|预处理模式串|||
|i=0,j=0|初始化|双指针从0开始|||
|while(i<n&&j<m)|主循环|i不回溯|||
|j==-1||t[i]==p[j]|匹配条件|j回溯到起点或字符相等|
|i++;j++|前进|匹配时双指针前进|||
|j=next[j]|跳转|不匹配时j跳转，i不动|||

```c
void buildNext(char *p, int *next) {
    int m = strlen(p);
    next[0] = -1;          // 第一个字符的next为-1
    int i = 0, j = -1;
    while (i < m - 1) {
        if (j == -1 || p[i] == p[j]) {
            i++; j++;
            next[i] = j;   // next[i+1] = j+1
        } else j = next[j]; // 回溯j
    }
}
```

|代码|功能|说明|||
|:--|:--|:--|---|---|
|next[0]=-1|第一个字符|-1表示退到起点|||
|i=0,j=-1|初始化|i指向当前，j指向前缀|||
|while(i<m-1)|循环|计算next[1]~next[m-1]|||
|j==-1||p[i]==p[j]|条件|可扩展前缀|
|next[i]=j|赋值|当前前缀长度|||
|j=next[j]|回溯|不匹配时j跳转|||

- **六、附加功能模块：BF与KMP性能对比**

---

|算法|最好情况|平均情况|最坏情况|额外空间|
|:--|:-:|:-:|:-:|:-:|
|BF|O(n)|O(n*m)|O(n*m)|O(1)|
|KMP|O(n+m)|O(n+m)|O(n+m)|O(m)|

BF 在匹配失败时，主串指针回溯到 i-j+1，之前比较过的字符可能被重复比较。KMP 通过 next 数组利用已匹配信息，每次匹配失败时，主串指针 i 不动，模式串指针 j 跳转到 next[j]，跳过已确认匹配的字符，避免重复比较。

- **十二、设计思路总结**

---

1. **BF朴素匹配**：最直观的匹配方式，外层遍历主串，内层遍历模式串。代码简单但最坏 O(n*m)。
    
2. **KMP算法**：通过 next 数组记录前缀信息，主串指针永不回溯，时间复杂度 O(n+m)。
    
3. **next数组**：每个位置的最长相同前后缀长度，递推求解。构建过程本身也使用了 KMP 思想。
    
4. **逐步演示**：通过分步输出帮助理解算法执行轨迹，看到 BF 的逐位比较和 KMP 的跳转。
    
5. **性能对比**：在相同大数据集上运行 10000 次取平均值，数据驱动地分析算法效率。
